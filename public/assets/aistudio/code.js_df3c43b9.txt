function doGet(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    ensureAllSheetsAndSchemas();
    const action = (e && e.parameter && e.parameter.action) ? String(e.parameter.action).trim() : "getInitialData";
    if (action === "setupAllSheets" || action === "ensureAllSheets") {
      return jsonResponse(setupSpreadsheetStructure());
    }
    const payload = getInitialSheetDataPayload(e && e.parameter ? e.parameter : {});
    return jsonResponse(payload);
  } catch (err) {
    return jsonResponse({
      status: "error",
      message: String(err && err.message ? err.message : err)
    });
  } finally {
    try { lock.releaseLock(); } catch (e2) {}
  }
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function cleanPhone(val) {
  if (val === undefined || val === null) return "";
  const s = String(val).replace(/^'+/, "").trim();
  if (!s) return "";
  const upper = s.toUpperCase();
  if (upper.indexOf("#ERROR") !== -1 || upper.indexOf("#VALUE") !== -1 || upper.indexOf("#NAME") !== -1) {
    return "";
  }
  if (s.indexOf("+") === 0) {
    const digits = s.substring(1).replace(/\D/g, "");
    if (!digits) return "";
    if (digits.length === 12 && digits.indexOf("91") === 0) {
      return "+91 " + digits.substring(2);
    }
    return "+" + digits;
  }
  const onlyDigits = s.replace(/\D/g, "");
  if (!onlyDigits) return "";
  if (onlyDigits.length === 12 && onlyDigits.indexOf("91") === 0) {
    return "+91 " + onlyDigits.substring(2);
  }
  return onlyDigits;
}

function toSheetSafeValue(val) {
  if (typeof val === "string") {
    const trimmed = val.trim();
    if (trimmed.indexOf("+") === 0) {
      return "'" + trimmed;
    }
  }
  return val;
}

function canonicalMonthlyCategory(cat) {
  const c = String(cat || "").trim().toLowerCase();
  if (!c) return "";
  if (c === "photo_1" || c === "photo1" || c === "electricity_meter" || c.indexOf("elec") !== -1) return "electricity_meter";
  if (c === "photo_2" || c === "photo2" || c === "completed_payment" || c.indexOf("pay") !== -1) return "completed_payment";
  if (c === "photo_3" || c === "photo3" || c === "water_meter_bill" || c.indexOf("water") !== -1) return "water_meter_bill";
  if (c === "photo_4" || c === "photo4" || c === "png_gas_meter" || c.indexOf("gas") !== -1 || c.indexOf("png") !== -1) return "png_gas_meter";
  return c;
}

function normKey(val) {
  return String(val || "").trim().toLowerCase();
}

function styleHeaderRange(sheet, numCols) {
  if (!sheet || numCols <= 0) return;
  try {
    const headerRange = sheet.getRange(1, 1, 1, numCols);
    headerRange
      .setFontWeight("bold")
      .setBackground("#e2e8f0")
      .setFontColor("#0f172a")
      .setHorizontalAlignment("left")
      .setVerticalAlignment("middle");
    sheet.setFrozenRows(1);
  } catch (e) {}
}

/**
 * Self-healing sheet creator & column schema validator:
 * - Creates any missing sheet tab automatically.
 * - Creates Row 1 with each field in its own separate column.
 * - Repairs missing or case-mismatched column headers without losing existing data.
 */
let _cachedSpreadsheet = null;
let _verifiedSheetCache = {};

function getActiveSpreadsheetFast() {
  if (!_cachedSpreadsheet) {
    _cachedSpreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  }
  return _cachedSpreadsheet;
}

function getOrCreateSheet(sheetName) {
  if (_verifiedSheetCache[sheetName]) {
    return _verifiedSheetCache[sheetName];
  }

  const ss = getActiveSpreadsheetFast();
  let sheet = ss.getSheetByName(sheetName);
  const expectedHeaders = DEFAULT_SCHEMAS[sheetName] || [];

  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    if (expectedHeaders.length > 0) {
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      styleHeaderRange(sheet, expectedHeaders.length);
    }
    if (sheetName === "Users") {
      seedDefaultAdminIfNeeded(sheet);
    }
    _verifiedSheetCache[sheetName] = sheet;
    return sheet;
  }

  // Self-Healing: Verify and restore separate column headers in Row 1
  const lastCol = sheet.getLastColumn();
  const lastRow = sheet.getLastRow();

  if (lastRow === 0 || lastCol === 0) {
    if (expectedHeaders.length > 0) {
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      styleHeaderRange(sheet, expectedHeaders.length);
    }
  } else {
    const rawHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    const currentHeaders = rawHeaders.map(function(h) {
      return String(h || "").trim();
    });
    const nonEmptyHeaders = currentHeaders.filter(Boolean);

    if (nonEmptyHeaders.length === 0 && expectedHeaders.length > 0) {
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      styleHeaderRange(sheet, expectedHeaders.length);
    } else if (expectedHeaders.length > 0) {
      const lowerToCanonical = {};
      expectedHeaders.forEach(function(eh) {
        lowerToCanonical[normKey(eh)] = eh;
      });

      let headerRowChanged = false;
      const normalizedCurrent = currentHeaders.map(function(ch) {
        const canonical = lowerToCanonical[normKey(ch)];
        if (canonical && canonical !== ch) {
          headerRowChanged = true;
          return canonical;
        }
        return ch;
      });

      if (headerRowChanged) {
        sheet.getRange(1, 1, 1, normalizedCurrent.length).setValues([normalizedCurrent]);
      }

      const existingLowerSet = {};
      normalizedCurrent.forEach(function(h) {
        if (h) existingLowerSet[normKey(h)] = true;
      });

      const missingHeaders = expectedHeaders.filter(function(eh) {
        return !existingLowerSet[normKey(eh)];
      });

      if (missingHeaders.length > 0) {
        const startCol = lastCol + 1;
        sheet.getRange(1, startCol, 1, missingHeaders.length).setValues([missingHeaders]);
        styleHeaderRange(sheet, lastCol + missingHeaders.length);
      }
    }
  }

  if (sheetName === "Users") {
    seedDefaultAdminIfNeeded(sheet);
  }

  _verifiedSheetCache[sheetName] = sheet;
  return sheet;
}

function ensureAllSheetsAndSchemas() {
  const ss = getActiveSpreadsheetFast();
  const sheetNames = Object.keys(DEFAULT_SCHEMAS);
  sheetNames.forEach(function(sName) {
    getOrCreateSheet(sName);
  });

  // Clean up default blank "Sheet1" if it is completely empty and all portal sheets exist
  try {
    const defaultSheet = ss.getSheetByName("Sheet1");
    if (defaultSheet && ss.getSheets().length > sheetNames.length) {
      if (defaultSheet.getLastRow() === 0 && defaultSheet.getLastColumn() === 0) {
        ss.deleteSheet(defaultSheet);
      }
    }
  } catch (e) {}
}

function readSheetObjects(sheetName) {
  const sheet = getOrCreateSheet(sheetName);
  const allValues = sheet.getDataRange().getValues();
  if (!allValues || allValues.length <= 1) return [];

  const headers = allValues[0].map(function(h) {
    return String(h || "").trim();
  });
  const results = [];

  for (let r = 1; r < allValues.length; r++) {
    const row = allValues[r];
    let hasNonEmptyCell = false;
    const obj = {};
    for (let idx = 0; idx < headers.length; idx++) {
      const header = headers[idx];
      if (header) {
        let val = row[idx];
        if (val instanceof Date) {
          val = val.toISOString();
        }
        if (val !== null && val !== undefined && String(val).trim() !== "") {
          hasNonEmptyCell = true;
        }
        obj[header] = (val !== null && val !== undefined) ? val : "";
      }
    }
    if (hasNonEmptyCell) {
      results.push(obj);
    }
  }
  return results;
}

function resolveRecordFieldForHeader(recordObj, headerName) {
  if (!recordObj || !headerName) return undefined;
  if (recordObj[headerName] !== undefined) return recordObj[headerName];
  const targetLower = normKey(headerName);
  const keys = Object.keys(recordObj);
  for (let i = 0; i < keys.length; i++) {
    if (normKey(keys[i]) === targetLower) {
      return recordObj[keys[i]];
    }
  }
  return undefined;
}

/**
 * Upserts a single record into its own separate row and maps each field into its
 * own separate column. If updating an existing row, preserves any existing column
 * values not explicitly overwritten in recordObj.
 */
function upsertRowByMatcher(sheetName, matcherFn, recordObj, forceAppendNewRow) {
  const sheet = getOrCreateSheet(sheetName);
  const lastCol = sheet.getLastColumn();
  if (lastCol === 0) return;

  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) {
    return String(h || "").trim();
  });

  const allValues = sheet.getDataRange().getValues();
  let targetRowNumber = -1;
  let existingRowObj = {};

  if (!forceAppendNewRow && allValues.length > 1) {
    for (let r = 1; r < allValues.length; r++) {
      const row = allValues[r];
      const obj = {};
      for (let c = 0; c < headers.length; c++) {
        const h = headers[c];
        if (h) {
          let v = row[c];
          if (v instanceof Date) v = v.toISOString();
          obj[h] = (v !== null && v !== undefined) ? v : "";
        }
      }
      if (matcherFn(obj)) {
        targetRowNumber = r + 1;
        existingRowObj = obj;
        break;
      }
    }
  }

  const rowValues = headers.map(function(h) {
    const incomingVal = resolveRecordFieldForHeader(recordObj, h);
    if (incomingVal !== undefined && incomingVal !== null) {
      if (typeof incomingVal === "object" && !(incomingVal instanceof Date)) {
        return JSON.stringify(incomingVal);
      }
      return toSheetSafeValue(incomingVal);
    }
    if (targetRowNumber > 0 && existingRowObj[h] !== undefined && existingRowObj[h] !== null) {
      return toSheetSafeValue(existingRowObj[h]);
    }
    return "";
  });

  if (targetRowNumber > 0) {
    sheet.getRange(targetRowNumber, 1, 1, headers.length).setValues([rowValues]);
  } else {
    sheet.appendRow(rowValues);
  }
}

function replaceAllSheetRows(sheetName, recordsArray) {
  const sheet = getOrCreateSheet(sheetName);
  const lastCol = sheet.getLastColumn();
  if (lastCol === 0) return;
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) {
    return String(h || "").trim();
  });

  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    sheet.getRange(2, 1, lastRow - 1, lastCol).clearContent();
  }

  if (!Array.isArray(recordsArray) || recordsArray.length === 0) return;

  const matrix = recordsArray.map(function(rec) {
    return headers.map(function(h) {
      const val = resolveRecordFieldForHeader(rec, h);
      if (val === undefined || val === null) return "";
      if (typeof val === "object" && !(val instanceof Date)) return JSON.stringify(val);
      return toSheetSafeValue(val);
    });
  });

  sheet.getRange(2, 1, matrix.length, headers.length).setValues(matrix);
}

function deleteRowsByMatcher(sheetName, matcherFn) {
  const sheet = getOrCreateSheet(sheetName);
  const allValues = sheet.getDataRange().getValues();
  if (!allValues || allValues.length <= 1) return;

  const headers = allValues[0].map(function(h) {
    return String(h || "").trim();
  });

  for (let r = allValues.length - 1; r >= 1; r--) {
    const row = allValues[r];
    const obj = {};
    for (let c = 0; c < headers.length; c++) {
      if (headers[c]) obj[headers[c]] = row[c];
    }
    if (matcherFn(obj)) {
      sheet.deleteRow(r + 1);
    }
  }
}

function updateRowsByMatcher(sheetName, matcherFn, updaterFn) {
  const sheet = getOrCreateSheet(sheetName);
  const allValues = sheet.getDataRange().getValues();
  if (!allValues || allValues.length <= 1) return;

  const headers = allValues[0].map(function(h) {
    return String(h || "").trim();
  });

  for (let r = 1; r < allValues.length; r++) {
    const row = allValues[r];
    const rowObj = {};
    for (let c = 0; c < headers.length; c++) {
      if (headers[c]) {
        let v = row[c];
        if (v instanceof Date) v = v.toISOString();
        rowObj[headers[c]] = (v !== null && v !== undefined) ? v : "";
      }
    }
    if (matcherFn(rowObj)) {
      const updatedPatch = updaterFn(rowObj) || {};
      const merged = Object.assign({}, rowObj, updatedPatch);
      const rowValues = headers.map(function(h) {
        const v = resolveRecordFieldForHeader(merged, h);
        if (v === undefined || v === null) return "";
        if (typeof v === "object" && !(v instanceof Date)) return JSON.stringify(v);
        return toSheetSafeValue(v);
      });
      sheet.getRange(r + 1, 1, 1, headers.length).setValues([rowValues]);
    }
  }
}

function resolveLogTimestampMs_(row) {
  if (!row || typeof row !== "object") return 0;
  const iso = String(row.UpdatedAt || row.Timestamp || row.LoginTime || "").trim();
  if (iso) {
    const parsed = Date.parse(iso);
    if (Number.isFinite(parsed) && parsed > 0) return parsed;
  }
  const hid = String(row.HistoryID || row.AuditID || row.id || "").trim();
  const m = hid.match(/(\d{10,13})/);
  if (m && m[1]) {
    const num = Number(m[1]);
    if (Number.isFinite(num) && num > 0) return num;
  }
  return 0;
}

function appendAndTrimLoginHistoryRow(entryObj, maxRows) {
  const limit = Number(maxRows) > 0 ? Number(maxRows) : 30;
  const existing = readSheetObjects("LoginHistory").filter(function(r) {
    return Boolean(r && (r.HistoryID || r.Username || r.LoginTime));
  });
  // Sort existing newest-first if possible, preserving sheet order ties
  const indexed = existing.map(function(row, idx) {
    return { row: row, idx: idx, ts: resolveLogTimestampMs_(row) };
  });
  indexed.sort(function(a, b) {
    if (b.ts !== a.ts) return b.ts - a.ts;
    return a.idx - b.idx;
  });
  const merged = (entryObj ? [entryObj] : []).concat(indexed.map(function(x) { return x.row; }));
  const trimmed = merged.slice(0, limit);
  replaceAllSheetRows("LoginHistory", trimmed);
  return trimmed;
}

function savePhotoToDriveFolderOnly(base64OrDataUrl, fileName, metadataObj) {
  const rawInput = String(base64OrDataUrl || "").trim();
  if (!rawInput) {
    throw new Error("No photo data provided for Google Drive upload.");
  }

  let mimeType = "image/jpeg";
  let base64Clean = rawInput;
  if (rawInput.indexOf("data:") === 0 && rawInput.indexOf("base64,") !== -1) {
    const parts = rawInput.split("base64,");
    const header = parts[0] || "";
    base64Clean = parts[1] || "";
    const mimeMatch = header.match(/data:([^;]+);/i);
    if (mimeMatch && mimeMatch[1]) {
      mimeType = mimeMatch[1].trim();
    }
  }

  base64Clean = String(base64Clean || "").replace(/\s+/g, "").trim();
  if (!base64Clean || base64Clean.indexOf("http") === 0) {
    throw new Error("Valid base64 photo data is required to upload to Google Drive.");
  }

  let decodedBytes;
  try {
    decodedBytes = Utilities.base64Decode(base64Clean);
  } catch (decodeErr) {
    throw new Error("Invalid base64 image payload for Google Drive: " + (decodeErr && decodeErr.message ? decodeErr.message : decodeErr));
  }
  if (!decodedBytes || decodedBytes.length === 0) {
    throw new Error("Decoded image payload is empty; cannot save to Google Drive.");
  }

  const folder = getPortalDrivePhotoFolder();
  const meta = Object.assign({}, metadataObj || {}, {
    driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
    uploadedAt: new Date().toISOString()
  });

  // Remove any older file in the Google Drive folder sharing the exact same slotKey so updates replace cleanly
  if (meta.slotKey) {
    try {
      const existingFiles = folder.getFiles();
      while (existingFiles.hasNext()) {
        const f = existingFiles.next();
        const desc = String(f.getDescription() || "").trim();
        let matchesSlot = false;
        if (desc && desc.charAt(0) === "{") {
          try {
            const parsedDesc = JSON.parse(desc);
            matchesSlot = Boolean(parsedDesc && parsedDesc.slotKey === meta.slotKey);
          } catch (parseErr) {
            matchesSlot = desc === meta.slotKey;
          }
        } else {
          matchesSlot = desc === meta.slotKey;
        }
        if (matchesSlot) {
          try { f.setTrashed(true); } catch (trashErr) {}
        }
      }
    } catch (iterErr) {}
  }

  let safeName = String(fileName || ("Photo_" + Date.now() + ".jpg")).trim();
  if (mimeType.toLowerCase().indexOf("png") !== -1) {
    safeName = safeName.replace(/\.(jpg|jpeg|webp)$/i, ".png");
    if (!/\.[a-z0-9]{2,5}$/i.test(safeName)) safeName += ".png";
  } else if (mimeType.toLowerCase().indexOf("webp") !== -1) {
    safeName = safeName.replace(/\.(jpg|jpeg|png)$/i, ".webp");
    if (!/\.[a-z0-9]{2,5}$/i.test(safeName)) safeName += ".webp";
  }
  const blob = Utilities.newBlob(decodedBytes, mimeType, safeName);
  const createdFile = folder.createFile(blob);
  createdFile.setDescription(JSON.stringify(meta));
  try {
    createdFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (shareErr) {
    try {
      createdFile.setSharing(DriveApp.Access.DOMAIN_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (domainErr) {}
  }

  const fileId = createdFile.getId();
  let driveWebUrl = "";
  try {
    driveWebUrl = String(createdFile.getUrl() || "").trim();
  } catch (urlErr) {}
  const shareableUrl = "https://drive.google.com/file/d/" + fileId + "/view?usp=sharing";
  const directViewUrl = "https://lh3.googleusercontent.com/d/" + fileId + "=s0";
  const previewUrl = (meta && meta.kind === "PORTAL_BACKGROUND")
    ? directViewUrl
    : ("https://lh3.googleusercontent.com/d/" + fileId + "=w1600");
  return {
    fileId: fileId,
    fileName: createdFile.getName(),
    shareableUrl: shareableUrl,
    driveUrl: driveWebUrl || shareableUrl,
    directViewUrl: directViewUrl,
    previewUrl: previewUrl,
    driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
    driveFolderUrl: GOOGLE_DRIVE_PHOTO_FOLDER_URL,
    uploadedAt: meta.uploadedAt,
    slotKey: meta.slotKey || ""
  };
}

function deletePhotoFromDriveFolderBySlot(slotKey) {
  if (!slotKey) return 0;
  let removedCount = 0;
  try {
    const folder = getPortalDrivePhotoFolder();
    const files = folder.getFiles();
    while (files.hasNext()) {
      const f = files.next();
      const desc = String(f.getDescription() || "").trim();
      let matchesSlot = false;
      if (desc && desc.charAt(0) === "{") {
        try {
          const parsedDesc = JSON.parse(desc);
          matchesSlot = Boolean(parsedDesc && parsedDesc.slotKey === slotKey);
        } catch (parseErr) {
          matchesSlot = desc === slotKey;
        }
      } else {
        matchesSlot = desc === slotKey;
      }
      if (matchesSlot) {
        try {
          f.setTrashed(true);
          removedCount++;
        } catch (e) {}
      }
    }
  } catch (err) {}
  return removedCount;
}

function getMonthlyEntryAadhaarKey(item, keyStr) {
  const itemAadh = String((item && (item.aadhaar || item.identityDocNumber)) || "").replace(/\D/g, "").slice(0, 12);
  if (itemAadh.length === 12) return "AADHAAR:" + itemAadh;
  const keyAadhMatch = String(keyStr || "").match(/(?:^|__)(\d{12})(?:__|$)/);
  if (keyAadhMatch && keyAadhMatch[1]) return "AADHAAR:" + keyAadhMatch[1];
  const rId = String((item && item.renterId) || "").trim().toLowerCase();
  if (rId) return "RENTERID:" + rId;
  const p = normKey(item && item.propertyName);
  const f = normKey(item && item.flatName);
  if (p && f && f !== "--") return "FLAT:" + p + "__" + f;
  return "";
}

function enforceMonthlyPhotoFifoInMap(monthlyPhotosMap, targetRenterObj, targetCategory, maxPerCat, trashDriveFiles) {
  const limit = maxPerCat || 3;
  const prunedKeys = [];
  if (!monthlyPhotosMap || typeof monthlyPhotosMap !== "object") return prunedKeys;

  const targetAadh = targetRenterObj ? String(targetRenterObj.aadhaar || targetRenterObj.identityDocNumber || "").replace(/\D/g, "").slice(0, 12) : "";
  const targetOwnerKey = targetAadh.length === 12
    ? ("AADHAAR:" + targetAadh)
    : (targetRenterObj ? getMonthlyEntryAadhaarKey(targetRenterObj, "") : "");
  const targetCatCanon = targetCategory ? canonicalMonthlyCategory(targetCategory) : "";

  const groups = {};
  Object.keys(monthlyPhotosMap).forEach(function(k) {
    const rec = monthlyPhotosMap[k];
    if (!rec || typeof rec !== "object") return;
    const ownerKey = getMonthlyEntryAadhaarKey(rec, k);
    const catCanon = canonicalMonthlyCategory(rec.category || "");
    if (!ownerKey || !catCanon) return;
    if (targetOwnerKey && ownerKey !== targetOwnerKey) return;
    if (targetCatCanon && catCanon !== targetCatCanon) return;
    const gKey = ownerKey + "::" + catCanon;
    if (!groups[gKey]) groups[gKey] = [];
    const monthNorm = String(rec.monthYear || rec.MonthYear || "").replace(/^'+/, "").trim();
    const ts = rec.uploadedAt ? (new Date(rec.uploadedAt).getTime() || 0) : 0;
    groups[gKey].push({
      key: k,
      rec: rec,
      month: monthNorm,
      ts: ts,
      fileId: String(rec.driveFileId || rec.fileId || "").trim()
    });
  });

  Object.keys(groups).forEach(function(gKey) {
    const items = groups[gKey];
    // Collapse alias keys belonging to the same (monthYear) into month buckets
    const monthBuckets = {};
    items.forEach(function(it) {
      const mBucket = it.month || ("NOMONTH_" + (it.fileId || it.key));
      if (!monthBuckets[mBucket]) {
        monthBuckets[mBucket] = { month: mBucket, maxTs: it.ts, entries: [] };
      }
      if (it.ts > monthBuckets[mBucket].maxTs) monthBuckets[mBucket].maxTs = it.ts;
      monthBuckets[mBucket].entries.push(it);
    });

    const sortedBuckets = Object.keys(monthBuckets).map(function(mk) { return monthBuckets[mk]; });
    sortedBuckets.sort(function(a, b) {
      if (a.month !== b.month) return a.month.localeCompare(b.month);
      return a.maxTs - b.maxTs;
    });

    while (sortedBuckets.length > limit) {
      const oldestBucket = sortedBuckets.shift();
      oldestBucket.entries.forEach(function(oldEntry) {
        if (trashDriveFiles) {
          if (oldEntry.fileId) {
            try { DriveApp.getFileById(oldEntry.fileId).setTrashed(true); } catch (e) {}
          }
          try { deletePhotoFromDriveFolderBySlot(oldEntry.key); } catch (e) {}
        }
        delete monthlyPhotosMap[oldEntry.key];
        prunedKeys.push(oldEntry.key);
      });
    }
  });

  return prunedKeys;
}

function hydratePhotosFromDriveFolder(renters, bills, khetRecords, globalConfig) {
  const monthlyPhotos = (globalConfig.renterMonthlyPhotos && typeof globalConfig.renterMonthlyPhotos === "object")
    ? globalConfig.renterMonthlyPhotos
    : {};
  try {
    const folder = getPortalDrivePhotoFolder();

    while (files.hasNext()) {
      const f = files.next();
      const desc = String(f.getDescription() || "").trim();
      if (!desc || desc.charAt(0) !== "{") continue;
      let meta = null;
      try {
        meta = JSON.parse(desc);
      } catch (e) {
        meta = null;
      }
      if (!meta || !meta.kind) continue;
      const fileId = f.getId();
      const previewUrl = (meta.kind === "PORTAL_BACKGROUND")
        ? ("https://lh3.googleusercontent.com/d/" + fileId + "=s0")
        : ("https://lh3.googleusercontent.com/d/" + fileId + "=w1600");

        const normF = normKey(meta.flatName || "");
        const normM = normKey(meta.monthYear || "");
        const normC = canonicalMonthlyCategory(meta.category || "");
        const flatSlotKey = normP && normF && normF !== "--" && normM && normC ? (normP + "__" + normF + "__" + normM + "__" + normC) : "";
        const aadhaarSlotKey = cleanAadhaar.length === 12 && normM && normC ? (cleanAadhaar + "__" + normM + "__" + normC) : "";
        const slotEntry = {
          slotKey: meta.slotKey,
          driveFileId: fileId,
          fileId: fileId,
          fileName: f.getName(),
          imageUrl: previewUrl,
          previewUrl: previewUrl,
          driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
          driveFolderUrl: GOOGLE_DRIVE_PHOTO_FOLDER_URL,
          propertyName: meta.propertyName || "",
          flatName: meta.flatName || "",
          monthYear: meta.monthYear || "",
          category: normC || meta.category || "",
          categoryLabel: meta.categoryLabel || "",
          renterId: meta.renterId || "",
          renterName: meta.renterName || "",
          username: meta.username || "",
          aadhaar: cleanAadhaar,
          identityDocNumber: cleanAadhaar,
          uploadedAt: meta.uploadedAt || ""
        };
        monthlyPhotos[meta.slotKey] = slotEntry;
        if (flatSlotKey && !monthlyPhotos[flatSlotKey]) monthlyPhotos[flatSlotKey] = slotEntry;
        if (aadhaarSlotKey && !monthlyPhotos[aadhaarSlotKey]) monthlyPhotos[aadhaarSlotKey] = slotEntry;
      } else if (meta.kind === "RENTER_DOCUMENT") {
        const dType = String(meta.docType || "").toLowerCase();
        const isIdentityBack = dType === "identity_back" || dType === "aadhaar_back";
        const isIdentity = dType === "identity" || dType === "identity_front";
        const metaAadhaar = String(meta.aadhaar || meta.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
        renters.forEach(function(r) {
          const matchById = meta.renterId && normKey(r.RenterID) === normKey(meta.renterId);
          const matchByFlat = meta.propertyName && meta.flatName &&
            normKey(r.PropertyName) === normKey(meta.propertyName) &&
            normKey(r.FlatName) === normKey(meta.flatName);
          const rAadhaar = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
          const matchByAadhaar = metaAadhaar.length === 12 && rAadhaar.length === 12 && metaAadhaar === rAadhaar;
          if (matchById || matchByFlat || matchByAadhaar) {
            if (isIdentityBack) {
              r.DriveIdentityBackFileId = fileId;
              r.DriveIdentityBackPreviewUrl = previewUrl;
              if (!r.IdentityBackImageURL) r.IdentityBackImageURL = previewUrl;
            } else if (isIdentity) {
              r.DriveIdentityFileId = fileId;
              r.DriveIdentityPreviewUrl = previewUrl;

              r.Photo2URL = r.Photo2URL || previewUrl;
            } else if (dType === "photo3" || dType === "photo_3") {
              r.Photo3URL = r.Photo3URL || previewUrl;
            } else if (dType === "photo4" || dType === "photo_4") {
              r.Photo4URL = r.Photo4URL || previewUrl;
            } else {
              r.DriveProfileFileId = fileId;
              r.DriveProfilePreviewUrl = previewUrl;
              if (!r.ProfileImageURL) r.ProfileImageURL = previewUrl;
            }
          }
        });
      } else if (meta.kind === "BILL_METER_PROOF" && meta.billId) {
        bills.forEach(function(b) {
          if (normKey(b.BillID) === normKey(meta.billId)) {
            b.DriveMeterProofFileId = fileId;
            b.DriveMeterProofPreviewUrl = previewUrl;
            if (!b.MeterProofUrl) b.MeterProofUrl = previewUrl;

            b.DrivePngProofFileId = fileId;
            b.DrivePngProofPreviewUrl = previewUrl;
            if (!b.PNGProofUrl) b.PNGProofUrl = previewUrl;
          }
        });

            b.DriveWaterProofFileId = fileId;
            b.DriveWaterProofPreviewUrl = previewUrl;
            if (!b.WaterProofUrl) b.WaterProofUrl = previewUrl;
          }
        });

        khetRecords.forEach(function(k) {
          if (normKey(k.KhetID || k.id) === normKey(meta.khetId)) {

        });
      } else if (meta.kind === "PORTAL_BACKGROUND") {

        if (!isCleared) {
          if (!globalConfig.bgDashboardUrl) globalConfig.bgDashboardUrl = previewUrl;
          if (!globalConfig.bgLoginUrl) globalConfig.bgLoginUrl = previewUrl;
          if (!globalConfig.backgroundUrl) globalConfig.backgroundUrl = previewUrl;
        }
      }
    }
  } catch (e) {}
  // Photos are kept permanently unless manually deleted by Admin
  globalConfig.renterMonthlyPhotos = monthlyPhotos;
  globalConfig.googleDriveFolderId = GOOGLE_DRIVE_PHOTO_FOLDER_ID;
  globalConfig.googleDriveFolderUrl = GOOGLE_DRIVE_PHOTO_FOLDER_URL;
}

function handleCheckUserLockout(data) {
  const cleanU = normKey(data.username || "");
  if (!cleanU) return { status: "ok", userExists: false };

  const usersSheet = getOrCreateSheet("Users");
  seedDefaultAdminIfNeeded(usersSheet);
  const users = readSheetObjects("Users");
  const match = users.find(function(u) { return normKey(u.Username) === cleanU; });

  if (!match) {
    return { status: "ok", userExists: false, userNotFound: true };
  }

  const st = normKey(match.Status || match.AccountStatus || "Active");
  const stage = Number(match.LockoutStage || 0);
  const permLockedByAttempts = (st === "blocked") && ((stage >= 4) || (String(match.IsPermanentlyBlocked || "").toLowerCase() === "true" && stage >= 4));
  if (permLockedByAttempts) {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: true,
      permanentlyLockedByAttempts: true,
      username: match.Username,
      message: "This account (@" + match.Username + ") has been blocked after repeated failed login attempts. Please contact the Administrator to unblock it."
    };
  }
  if (st === "blocked") {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: false,
      permanentlyLockedByAttempts: false,
      username: match.Username,
      message: "This user account (@" + match.Username + ") has been blocked by an administrator."
    };
  }

  const lockedUntil = Number(match.LockedUntil || 0);
  const attempts = Number(match.FailedAttempts || 0);
  const tierLabel = String(match.LockedTierLabel || "1 minute");

  if (lockedUntil > Date.now()) {
    const remMs = lockedUntil - Date.now();
    return {
      status: "locked",
      userExists: true,
      locked: true,
      username: match.Username,
      lockedUntil: lockedUntil,
      remainingMinutes: Math.max(1, Math.ceil(remMs / 60000)),
      remainingSeconds: Math.max(1, Math.ceil(remMs / 1000)),
      tier: stage || 1,
      tierLabel: tierLabel,
      message: "Username @" + match.Username + " is locked for " + tierLabel + "."
    };
  }

  return {
    status: "ok",
    userExists: true,
    locked: false,
    username: match.Username,
    tier: stage,
    attempts: attempts
  };
}

function handleRecordFailedAttempt(data) {
  const cleanU = normKey(data.username || "");
  if (!cleanU) return { status: "error", userExists: false };

  const users = readSheetObjects("Users");
  const match = users.find(function(u) { return normKey(u.Username) === cleanU; });
  if (!match) {
    return { status: "ok", userExists: false, userNotFound: true };
  }

  const now = Date.now();
  const nowIso = new Date().toISOString();
  let stage = Number(match.LockoutStage || 0);
  let attempts = Number(match.FailedAttempts || 0) + 1;
  let lockedUntil = Number(match.LockedUntil || 0);
  let tierLabel = String(match.LockedTierLabel || "");
  let isPermBlocked = (normKey(match.Status || match.AccountStatus || "Active") === "blocked") && ((stage >= 4) || (String(match.IsPermanentlyBlocked || "").toLowerCase() === "true" && stage >= 4));

  if (isPermBlocked) {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: true,
      permanentlyLockedByAttempts: true,
      username: match.Username,
      message: "Account @" + match.Username + " is blocked due to repeated failed attempts. Contact an Administrator to unblock it."
    };
  }

  if (normKey(match.Status) === "blocked") {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: false,
      username: match.Username,
      message: "Account @" + match.Username + " is blocked by an administrator."
    };
  }

  if (attempts >= 3) {
    attempts = 0;
    if (stage === 0) {
      stage = 1;
      lockedUntil = now + 60 * 1000;
      tierLabel = "1 Minute";
    } else if (stage === 1) {
      stage = 2;
      lockedUntil = now + 60 * 60 * 1000;
      tierLabel = "1 Hour";
    } else if (stage === 2) {
      stage = 3;
      lockedUntil = now + 24 * 60 * 60 * 1000;
      tierLabel = "24 Hours";
    } else {
      stage = 4;
      lockedUntil = 0;
      tierLabel = "Permanent Block";
      isPermBlocked = true;
    }
  }

  updateRowsByMatcher(
    "Users",
    function(r) { return normKey(r.Username) === cleanU; },
    function(prev) {
      return {
        Status: isPermBlocked ? "Blocked" : (prev.Status || "Active"),
        AccountStatus: isPermBlocked ? "Blocked" : (prev.AccountStatus || "Active"),
        ActiveSessionToken: isPermBlocked ? "" : (prev.ActiveSessionToken || ""),
        LockoutStage: stage,
        FailedAttempts: attempts,
        LockedUntil: lockedUntil,
        LockedTierLabel: tierLabel,
        IsPermanentlyBlocked: isPermBlocked ? "true" : "false",
        UpdatedAt: nowIso
      };
    }
  );

  if (isPermBlocked) {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: true,
      permanentlyLockedByAttempts: true,
      username: match.Username,
      message: "User @" + match.Username + " has been blocked after 4 stages of failed attempts. Contact an Administrator to unblock this account."
    };
  }

  if (lockedUntil > now) {
    const remMs = lockedUntil - now;
    return {
      status: "locked",
      userExists: true,
      locked: true,
      username: match.Username,
      tier: stage,
      tierLabel: tierLabel,
      lockedUntil: lockedUntil,
      remainingMinutes: Math.max(1, Math.ceil(remMs / 60000)),
      remainingSeconds: Math.max(1, Math.ceil(remMs / 1000)),
      message: "Username @" + match.Username + " is locked for " + tierLabel + "."
    };
  }

  return {
    status: "attempt_counted",
    userExists: true,
    locked: false,
    username: match.Username,
    tier: stage,
    attempts: attempts,
    attemptsRemaining: Math.max(0, 3 - attempts)
  };
}

function handleLoginUser(data) {
  const enteredUser = String(data.username || "").trim();
  const enteredPass = String(data.password || "").trim();
  const sessionToken = String(data.sessionToken || ("SESS-" + Date.now())).trim();
  const nowIso = new Date().toISOString();

  if (!enteredUser || !enteredPass) {
    return { status: "error", message: "Username and password are required." };
  }

  const usersSheet = getOrCreateSheet("Users");
  seedDefaultAdminIfNeeded(usersSheet);
  const users = readSheetObjects("Users").filter(function(u) {
    return Boolean(String(u.Username || "").trim());
  });
  const lowerUser = enteredUser.toLowerCase();

  const userMatch = users.find(function(u) {
    return normKey(u.Username) === lowerUser;
  });

  if (!userMatch) {
    return { status: "error", userExists: false, userNotFound: true, message: "User ID not found in Google Sheets. Please verify your credentials." };
  }

  const userStatus = String(userMatch.Status || userMatch.AccountStatus || "Active").trim();
  const matchStage = Number(userMatch.LockoutStage || 0);
  const isPermLocked = (userStatus.toLowerCase() === "blocked") && ((matchStage >= 4) || (String(userMatch.IsPermanentlyBlocked || "").toLowerCase() === "true" && matchStage >= 4));
  if (isPermLocked) {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: true,
      permanentlyLockedByAttempts: true,
      message: "This account has been blocked after repeated failed login attempts. Please contact the Administrator to unblock it."
    };
  }

  if (userStatus.toLowerCase() === "blocked") {
    return {
      status: "blocked",
      userExists: true,
      isPermanentlyBlocked: false,
      permanentlyLockedByAttempts: false,
      message: "This account has been blocked by an administrator. Login access is strictly denied."
    };
  }

  const lockedUntilMs = Number(userMatch.LockedUntil || 0);
  if (lockedUntilMs > Date.now()) {
    const remMs = lockedUntilMs - Date.now();
    return {
      status: "locked",
      userExists: true,
      locked: true,
      username: userMatch.Username,
      lockedUntil: lockedUntilMs,
      remainingMinutes: Math.max(1, Math.ceil(remMs / 60000)),
      remainingSeconds: Math.max(1, Math.ceil(remMs / 1000)),

    if (lockRow && lockRow.ConfigValueJSON) {
      const parsedLocks = JSON.parse(lockRow.ConfigValueJSON) || {};
      if (parsedLocks[lowerUser]) {
        delete parsedLocks[lowerUser];
        handleSaveConfigItem("loginLockouts", parsedLocks);
      }
    }
  } catch (e) {}

  // Prepend structured row to LoginHistory sheet and auto-delete oldest beyond 30 records (Date, Month, Year only)
  appendAndTrimLoginHistoryRow({
    HistoryID: "LH-" + Date.now(),
    Username: userMatch.Username,
    Role: userMatch.Role || "User",
    LoginTime: formatDateOnlyDDMMMYYYY_(data.loginTime || new Date()),
    DeviceInfo: String(data.deviceInfo || "Web Browser").slice(0, 150),
    Status: "Success",
    SessionToken: sessionToken,
    UpdatedAt: nowIso
  }, 30);

  const dbPayload = getInitialSheetDataPayload();
  const userObj = {
    Username: userMatch.Username,
    Password: storedPass,
    FullName: userMatch.FullName || userMatch.Username,
    Role: userMatch.Role || "Viewer",

    AccountStatus: "Active",
    MobileNumber: userMatch.MobileNumber || "",

    AssignedFlat: userMatch.AssignedFlat || "",
    AssignedAadhaar: userMatch.AssignedAadhaar || userMatch.AadhaarNumber || "",
    ManagerPermissions: userMatch.ManagerPermissions || "",
    Theme: userMatch.Theme || "system",
    BgLogin: userMatch.BgLogin || "",
    BgDashboard: userMatch.BgDashboard || "",
    ActiveSessionToken: sessionToken
  };

  return {
    status: "success",
    username: userObj.Username,
    fullName: userObj.FullName,
    mobile: userObj.MobileNumber,
    role: userObj.Role,
    theme: userObj.Theme,

    bgDashboardUrl: (dbPayload.globalConfig && dbPayload.globalConfig.bgDashboardUrl) || userObj.BgDashboard || "",
    sessionToken: sessionToken,
    user: userObj,
    dbPayload: dbPayload,
    properties: dbPayload.properties,
    flats: dbPayload.flats,
    renters: dbPayload.renters,

    return { status: "error", message: "Username must be at least 3 characters long." };
  }
  const newLower = normKey(newUsernameClean);
  const nowIso = new Date().toISOString();

  const users = readSheetObjects("Users");
  if (newLower !== oldLower) {
    const dup = users.find(function(u) { return normKey(u.Username) === newLower; });
    if (dup) return { status: "duplicate", message: "This username is already taken." };
  }

  const existing = users.find(function(u) { return normKey(u.Username) === oldLower; }) || {};
  const nextStatus = String(data.newStatus || data.status || existing.Status || "Active").trim();
  const isBlockedNow = nextStatus.toLowerCase() === "blocked";

  if (requesterLower && requesterLower === oldLower && isBlockedNow) {

      status: "error",
      message: "You cannot block your own currently logged-in account. Log in with another Admin account to block this account."
    };
  }

  const nextRole = String(data.role || data.newRole || existing.Role || "Viewer").trim();
  const nextPassword = (data.newPassword !== undefined && String(data.newPassword).trim() !== "")
    ? String(data.newPassword).trim()
    : String(existing.Password || "").trim();
  const passwordChanged = nextPassword !== String(existing.Password || "").trim();
  const rawAadh = data.assignedAadhaar !== undefined
    ? String(data.assignedAadhaar || "").replace(/\D/g, "").slice(0, 12)
    : String(existing.AssignedAadhaar || existing.AadhaarNumber || "");

  const updatedObj = {

    Password: nextPassword,
    Role: nextRole,
    Status: nextStatus,

    AutoSeededDefault: "false",
    LockoutStage: isBlockedNow ? Number(existing.LockoutStage || 0) : 0,

function handleDeleteUser(data) {
  const targetUser = String(data.targetUsername || "").trim();
  const targetLower = normKey(targetUser);
  const requesterLower = normKey(data.requesterUsername || "");

  if (requesterLower && requesterLower === targetLower) {
    return {
      status: "error",
      message: "You cannot delete your own currently logged-in account."
    };
  }

  const existingUsers = readSheetObjects("Users");
  const uRecord = existingUsers.find(function(r) { return normKey(r.Username) === targetLower; }) || {};
  const isRenter = normKey(uRecord.Role || data.targetRole || "") === "renter";
  const assignedProp = String(uRecord.AssignedProperty || data.assignedProperty || "").trim();
  const assignedFlat = String(uRecord.AssignedFlat || data.assignedFlat || "").trim();

  deleteRowsByMatcher("Users", function(r) { return normKey(r.Username) === targetLower; });

  // Also automatically delete all MaintenanceTickets submitted by this deleted renter
  const matchesDeletedRenterTicket = function(t) {
    if (!t) return false;
    const tUser = normKey(t.Username || t.username || "");
    if (tUser && tUser === targetLower) return true;
    if (isRenter && assignedProp && assignedFlat) {
      const tProp = normKey(t.PropertyName || t.propertyName || "");
      const tFlat = normKey(t.FlatName || t.flatName || "");
      if (tProp === normKey(assignedProp) && tFlat === normKey(assignedFlat)) return true;
    }
    return false;
  };

  deleteRowsByMatcher("MaintenanceTickets", matchesDeletedRenterTicket);

  let filteredTickets = [];
  try {
    const rawCfgRows = readSheetObjects("GlobalConfig");
    const foundTicketsCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "maintenancetickets"; });
    let existingTickets = [];
    if (foundTicketsCfg && foundTicketsCfg.ConfigValueJSON) {
      existingTickets = JSON.parse(foundTicketsCfg.ConfigValueJSON) || [];
    }
    if (!Array.isArray(existingTickets)) existingTickets = [];
    filteredTickets = existingTickets.filter(function(t) { return !matchesDeletedRenterTicket(t); });
    handleSaveConfigItem("maintenanceTickets", filteredTickets);
  } catch (e) {}

  const usersSheet = getOrCreateSheet("Users");
  seedDefaultAdminIfNeeded(usersSheet);

  return {
    status: "success",
    users: readSheetObjects("Users").filter(function(u) { return Boolean(u.Username); }),
    maintenanceTickets: filteredTickets,
    message: "User @" + targetUser + " permanently deleted from Google Sheets."
  };
}

function handleChangeUsername(data) {
  const oldU = normKey(data.currentUsernameInput || data.loggedUsername || data.oldUsername || "");

  const newLower = normKey(newU);
  const nowIso = new Date().toISOString();

  if (!newU || newU.length < 3) return { status: "error", message: "Username must be at least 3 characters long." };

  const users = readSheetObjects("Users");
  if (oldU !== newLower) {
    const dup = users.find(function(r) { return normKey(r.Username) === newLower; });
    if (dup) return { status: "duplicate", message: "This username is already taken." };
  }

  const match = users.find(function(r) { return normKey(r.Username) === oldU; }) || {};
  deleteRowsByMatcher("Users", function(r) { return normKey(r.Username) === oldU; });

  const updatedUser = Object.assign({}, match, {
    Username: newU,
    AutoSeededDefault: "false",
    UpdatedAt: nowIso
  });
  upsertRowByMatcher("Users", function(r) { return normKey(r.Username) === newLower; }, updatedUser);

    status: "success",
    newUsername: newU,
    users: readSheetObjects("Users").filter(function(u) { return Boolean(u.Username); }),

  };

function findMatchingUserByAadhaarAndEmail(cleanAadhaar, cleanEmail) {

    });
  }) || null;
}

function formatDateOnlyDDMMMYYYY_(rawDate) {
  var months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  try {
    if (!rawDate) {
      return Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "Asia/Kolkata", "dd MMM yyyy");
    }
    if (Object.prototype.toString.call(rawDate) === "[object Date]") {
      return Utilities.formatDate(rawDate, Session.getScriptTimeZone() || "Asia/Kolkata", "dd MMM yyyy");
    }
    var s = String(rawDate).trim().replace(/^'+/, "");
    var dmy = s.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})/);
    if (dmy) {
      var mIdx = parseInt(dmy[2], 10) - 1;
      if (mIdx >= 0 && mIdx <= 11) {
        return ("0" + parseInt(dmy[1], 10)).slice(-2) + " " + months[mIdx] + " " + dmy[3];
      }
    }
    var ymd = s.match(/^(\d{4})[/-](\d{1,2})[/-](\d{1,2})/);
    if (ymd) {
      var mIdx2 = parseInt(ymd[2], 10) - 1;
      if (mIdx2 >= 0 && mIdx2 <= 11) {
        return ("0" + parseInt(ymd[3], 10)).slice(-2) + " " + months[mIdx2] + " " + ymd[1];
      }
    }
    var dMonY = s.match(/^(\d{1,2})[\s-]+([A-Za-z]{3,9})[\s-,]+(\d{4})/);
    if (dMonY) {
      var shortM = dMonY[2].slice(0, 3).toLowerCase();
      for (var i = 0; i < months.length; i++) {
        if (months[i].toLowerCase() === shortM) {
          return ("0" + parseInt(dMonY[1], 10)).slice(-2) + " " + months[i] + " " + dMonY[3];
        }
      }
    }
    var dt = new Date(s);
    if (!isNaN(dt.getTime())) {
      return Utilities.formatDate(dt, Session.getScriptTimeZone() || "Asia/Kolkata", "dd MMM yyyy");
    }
  } catch (e) {}
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "Asia/Kolkata", "dd MMM yyyy");
}

function handleEditProperty(data) {
  const oldName = String(data.oldName || data.oldPropertyName || data.propertyName || "").trim();
  const newName = String(data.newName || data.newPropertyName || oldName).trim();
  if (!oldName || !newName) return { status: "error", message: "Old and new names are required." };
  const nowIso = new Date().toISOString();
  const contactMob = cleanPhone(data.contractNumber || data.contractMobile || "");
  const hasPropAddr = data.propertyAddress !== undefined || data.propertyLocationAddress !== undefined;

  updateRowsByMatcher(

function extractDriveFileIdFromUrl_(rawUrl) {
  const s = String(rawUrl || "").trim();
  if (!s) return "";
  let m = s.match(/\/file\/d\/([a-zA-Z0-9_-]{15,})/);
  if (m && m[1]) return m[1];
  m = s.match(/\/d\/([a-zA-Z0-9_-]{15,})/);
  if (m && m[1]) return m[1];
  m = s.match(/[?&]id=([a-zA-Z0-9_-]{15,})/);
  if (m && m[1]) return m[1];
  if (/^[a-zA-Z0-9_-]{20,}$/.test(s)) return s;
  return "";
}

function addDriveUrlOrIdToSet_(targetSet, rawVal) {
  const s = String(rawVal || "").trim();
  if (!s || s.indexOf("data:image/svg+xml") === 0) return;
  const fId = extractDriveFileIdFromUrl_(s);
  if (fId) targetSet[fId] = true;
}

function cleanupPropertyOrFlatAssetsAndConfig_(pName, fNameOpt) {
  const cleanP = normKey(pName);
  const cleanF = normKey(fNameOpt || "");
  const isWholeProperty = !cleanF;

  const matchesScope = function(rowProp, rowFlat) {
    if (normKey(rowProp) !== cleanP) return false;
    if (isWholeProperty) return true;
    return normKey(rowFlat) === cleanF;
  };

  const fileIdsToTrash = {};
  const renterIdsToDelete = {};
  const billIdsToDelete = {};
  const deletedAadhaars = {};
  const remainingAadhaars = {};

  // 1. Collect all Renter photos & IDs BEFORE deleting rows
  const allRenters = readSheetObjects("Renters");
  allRenters.forEach(function(r) {
    const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
    if (matchesScope(r.PropertyName, r.FlatName)) {
      if (r.RenterID) renterIdsToDelete[normKey(r.RenterID)] = true;
      if (rAadh.length === 12) deletedAadhaars[rAadh] = true;
      [
        r.ProfileImageURL, r.IdentityImageURL, r.IdentityBackImageURL,
        r.DriveProfilePreviewUrl, r.DriveIdentityPreviewUrl, r.DriveIdentityBackPreviewUrl,
        r.Photo1URL, r.Photo2URL, r.Photo3URL, r.Photo4URL
      ].forEach(function(u) { addDriveUrlOrIdToSet_(fileIdsToTrash, u); });
    } else if (rAadh.length === 12) {
      remainingAadhaars[rAadh] = true;
    }
  });

  // 2. Collect all Bill meter/utility proofs & Bill IDs BEFORE deleting rows
  const allBills = readSheetObjects("Bills");
  allBills.forEach(function(b) {
    if (matchesScope(b.PropertyName, b.FlatName)) {
      if (b.BillID) billIdsToDelete[normKey(b.BillID)] = true;
      [
        b.MeterProofUrl, b.PNGProofUrl, b.WaterProofUrl,
        b.DriveMeterProofPreviewUrl, b.DrivePngProofPreviewUrl, b.DriveWaterProofPreviewUrl
      ].forEach(function(u) { addDriveUrlOrIdToSet_(fileIdsToTrash, u); });
    }
  });

  // 3. Collect Renter Usernames tied strictly to this deleted scope so chats/users are also cleaned
  const allUsers = readSheetObjects("Users");
  const deletedUsernames = [];
  const deletedUsernamesMap = {};
  allUsers.forEach(function(u) {
    if (normKey(u.Role) !== "renter") return;
    const uProp = normKey(u.AssignedProperty);
    const uFlat = normKey(u.AssignedFlat);
    const uAadh = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
    const matchDirect = isWholeProperty ? (uProp === cleanP) : (uProp === cleanP && uFlat === cleanF);
    const matchOrphanAadhaar = uAadh.length === 12 && Boolean(deletedAadhaars[uAadh]) && !remainingAadhaars[uAadh];
    if (matchDirect || matchOrphanAadhaar) {
      const uname = String(u.Username || "").trim();
      if (uname) {
        deletedUsernames.push(uname);
        deletedUsernamesMap[normKey(uname)] = true;
      }
    }
  });

  // 4. Clean GlobalConfig collections & collect any Drive files referenced there
  const rawCfgRows = readSheetObjects("GlobalConfig");
  const cfgByKey = {};
  rawCfgRows.forEach(function(r) {
    if (r.ConfigKey && r.ConfigValueJSON) {
      try { cfgByKey[normKey(r.ConfigKey)] = { key: r.ConfigKey, val: JSON.parse(r.ConfigValueJSON) }; } catch (e) {}
    }
  });

  if (cfgByKey["rentermonthlyphotos"] && cfgByKey["rentermonthlyphotos"].val && typeof cfgByKey["rentermonthlyphotos"].val === "object") {
    const photosMap = cfgByKey["rentermonthlyphotos"].val;
    let changed = false;
    Object.keys(photosMap).forEach(function(k) {
      const entry = photosMap[k];
      if (!entry || typeof entry !== "object") return;
      const eP = normKey(entry.propertyName);
      const eF = normKey(entry.flatName);
      const eRId = normKey(entry.renterId);
      const eAadh = String(entry.aadhaar || entry.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
      const matchEntry = (eP && matchesScope(entry.propertyName, entry.flatName)) ||
        (eRId && Boolean(renterIdsToDelete[eRId])) ||
        (eAadh.length === 12 && Boolean(deletedAadhaars[eAadh]) && !remainingAadhaars[eAadh]) ||
        (isWholeProperty ? (k.indexOf(cleanP + "__") !== -1) : (k.indexOf(cleanP + "__" + cleanF + "__") !== -1));
      if (matchEntry) {
        addDriveUrlOrIdToSet_(fileIdsToTrash, entry.fileId || entry.driveFileId);
        addDriveUrlOrIdToSet_(fileIdsToTrash, entry.imageUrl);
        addDriveUrlOrIdToSet_(fileIdsToTrash, entry.previewUrl);
        addDriveUrlOrIdToSet_(fileIdsToTrash, entry.shareableUrl);
        delete photosMap[k];
        changed = true;
      }
    });
    if (changed) handleSaveConfigItem("renterMonthlyPhotos", photosMap);
  }

  if (cfgByKey["screenshotrequests"] && cfgByKey["screenshotrequests"].val) {
    const srVal = cfgByKey["screenshotrequests"].val;
    if (Array.isArray(srVal)) {
      const filtered = srVal.filter(function(req) {
        if (!req || typeof req !== "object") return true;
        const rId = normKey(req.renterId);
        return !(matchesScope(req.propertyName, req.flatName) || (rId && Boolean(renterIdsToDelete[rId])));
      });
      if (filtered.length !== srVal.length) handleSaveConfigItem("screenshotRequests", filtered);
    } else if (typeof srVal === "object") {
      let changedSr = false;
      Object.keys(srVal).forEach(function(k) {
        const req = srVal[k];
        if (!req || typeof req !== "object") return;
        const rId = normKey(req.renterId);
        const rAadh = String(req.aadhaar || req.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
        const hit = matchesScope(req.propertyName, req.flatName) ||
          (rId && Boolean(renterIdsToDelete[rId])) ||
          (rAadh.length === 12 && Boolean(deletedAadhaars[rAadh]) && !remainingAadhaars[rAadh]);
        if (hit) {
          delete srVal[k];
          changedSr = true;
        }
      });
      if (changedSr) handleSaveConfigItem("screenshotRequests", srVal);
    }
  }

  if (cfgByKey["paymentproofs"] && Array.isArray(cfgByKey["paymentproofs"].val)) {
    const ppArr = cfgByKey["paymentproofs"].val;
    const filteredPp = ppArr.filter(function(item) {
      if (!item || typeof item !== "object") return true;
      const hit = matchesScope(item.propertyName, item.flatName) ||
        (item.renterId && Boolean(renterIdsToDelete[normKey(item.renterId)])) ||
        (item.billId && Boolean(billIdsToDelete[normKey(item.billId)]));
      if (hit) {
        addDriveUrlOrIdToSet_(fileIdsToTrash, item.fileId || item.driveFileId);
        addDriveUrlOrIdToSet_(fileIdsToTrash, item.proofUrl || item.imageUrl || item.screenshotUrl);
        return false;
      }
      return true;
    });
    if (filteredPp.length !== ppArr.length) handleSaveConfigItem("paymentProofs", filteredPp);
  }

  if (cfgByKey["maintenancetickets"] && Array.isArray(cfgByKey["maintenancetickets"].val)) {
    const mtArr = cfgByKey["maintenancetickets"].val;
    const filteredMt = mtArr.filter(function(item) {
      if (!item || typeof item !== "object") return true;
      const hit = matchesScope(item.propertyName, item.flatName) ||
        (item.username && Boolean(deletedUsernamesMap[normKey(item.username)]));
      if (hit) {
        addDriveUrlOrIdToSet_(fileIdsToTrash, item.fileId || item.driveFileId);
        addDriveUrlOrIdToSet_(fileIdsToTrash, item.photoUrl || item.imageUrl);
        return false;
      }
      return true;
    });
    if (filteredMt.length !== mtArr.length) handleSaveConfigItem("maintenanceTickets", filteredMt);
  }

  if (cfgByKey["privatechats"] && cfgByKey["privatechats"].val && typeof cfgByKey["privatechats"].val === "object") {
    const chatsObj = cfgByKey["privatechats"].val;
    let changedChats = false;
    Object.keys(chatsObj).forEach(function(chKey) {
      const chNorm = normKey(chKey);
      const matchChatUser = Object.keys(deletedUsernamesMap).some(function(uN) {
        return uN && (chNorm === uN || chNorm.indexOf(uN + "::") !== -1 || chNorm.indexOf("::" + uN) !== -1 || chNorm.indexOf("__" + uN) !== -1);
      });
      if (matchChatUser) {
        delete chatsObj[chKey];
        changedChats = true;
      }
    });
    if (changedChats) handleSaveConfigItem("privateChats", chatsObj);
  }

  // 5. Trash every collected Google Drive file ID & scan Portal Drive Folder metadata for any remaining files of this property/flat
  let trashedDriveFiles = 0;
  Object.keys(fileIdsToTrash).forEach(function(fid) {
    try {
      const f = DriveApp.getFileById(fid);
      if (f && !f.isTrashed()) {
        f.setTrashed(true);
        trashedDriveFiles++;
      }
    } catch (e) {}
  });

  try {
    const folder = getPortalDrivePhotoFolder();
    const files = folder.getFiles();
    while (files.hasNext()) {
      const f = files.next();
      const desc = String(f.getDescription() || "").trim();
      let meta = null;
      if (desc && desc.charAt(0) === "{") {
        try { meta = JSON.parse(desc); } catch (e) { meta = null; }
      }
      if (!meta || typeof meta !== "object") continue;
      if (meta.kind === "PORTAL_BACKGROUND") continue;

      const mProp = meta.propertyName || "";
      const mFlat = meta.flatName || "";
      const mRId = normKey(meta.renterId || "");
      const mBId = normKey(meta.billId || "");
      const mSlot = String(meta.slotKey || "").toLowerCase();
      const mAadh = String(meta.aadhaar || meta.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);

      const hitByMeta = (mProp && matchesScope(mProp, mFlat)) ||
        (mRId && Boolean(renterIdsToDelete[mRId])) ||
        (mBId && Boolean(billIdsToDelete[mBId])) ||
        (mAadh.length === 12 && Boolean(deletedAadhaars[mAadh]) && !remainingAadhaars[mAadh]) ||
        (mSlot && (isWholeProperty ? (mSlot.indexOf("__" + cleanP + "__") !== -1) : (mSlot.indexOf("__" + cleanP + "__" + cleanF + "__") !== -1)));

      if (hitByMeta) {
        try {
          if (!f.isTrashed()) {
            f.setTrashed(true);
            trashedDriveFiles++;
          }
        } catch (trashErr) {}
      }
    }
  } catch (driveErr) {}

  // 6. Delete linked Renter user rows from Users sheet
  if (deletedUsernames.length > 0) {
    deleteRowsByMatcher("Users", function(u) {
      return Boolean(u.Username && deletedUsernamesMap[normKey(u.Username)]);
    });
  }

  return {
    trashedDriveFiles: trashedDriveFiles,
    deletedUsernames: deletedUsernames
  };
}

          BathroomCount: nextBathroomCount,
          ToiletCount: nextToiletCount,
          RentAmount: finalRentAmount > 0 ? finalRentAmount : (parseFloat(prevFlat.RentAmount) || 0),
          ElecMeterNumber: record.ElecMeterNumber || prevFlat.ElecMeterNumber || "",
          ElecRate: elecRateVal !== "" ? elecRateVal : prevFlat.ElecRate,
          ElecFixedRate: elecRateVal !== "" ? elecRateVal : prevFlat.ElecFixedRate,
          PNGNumber: record.PNGNumber || prevFlat.PNGNumber || "",
          PNGMeterNumber: record.PNGMeterNumber || prevFlat.PNGMeterNumber || "",
          PNGRate: pngRateVal !== "" ? pngRateVal : prevFlat.PNGRate,
          PNGFixedRate: pngRateVal !== "" ? pngRateVal : prevFlat.PNGFixedRate,
          WaterNumber: record.WaterNumber || prevFlat.WaterNumber || "",
          WaterMeterNumber: record.WaterMeterNumber || prevFlat.WaterMeterNumber || "",
          WaterRate: waterRateVal !== "" ? waterRateVal : prevFlat.WaterRate,
          WaterFixedRate: waterRateVal !== "" ? waterRateVal : prevFlat.WaterFixedRate,
          SecurityDeposit: record.SecurityDeposit > 0 ? record.SecurityDeposit : (prevFlat.SecurityDeposit || 0),
          DepositDate: record.DepositDate || prevFlat.DepositDate || "",
          DepositMode: record.DepositMode || prevFlat.DepositMode || "UPI",
          EscalationPct: record.EscalationPct !== undefined ? record.EscalationPct : (prevFlat.EscalationPct || 10),
          UpdatedAt: nowIso
        };
      }
    );
  }

  if (record.PropertyName && hasProfilePropAddr && profilePropAddrVal !== "") {
    updateRowsByMatcher(
      "Properties",
      function(row) { return normKey(row.PropertyName) === normKey(record.PropertyName); },
      function() { return { PropertyAddress: profilePropAddrVal, UpdatedAt: nowIso }; }
    );
    updateRowsByMatcher(
      "Flats",
      function(f) { return normKey(f.PropertyName) === normKey(record.PropertyName); },
      function() { return { PropertyAddress: profilePropAddrVal, UpdatedAt: nowIso }; }
    );
  }

  // Sync personal KYC details (NEVER overwriting RentAmount, PropertyName, or FlatName!) across all stays sharing the same 12-digit Aadhaar
  const prevAadhaarDigits = String(data.previousAadhaar || prevRow.IdentityDocNumber || prevRow.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
  if (cleanAadhDigits.length === 12 || prevAadhaarDigits.length === 12) {
    updateRowsByMatcher(
      "Renters",
      function(r) {
        if (normKey(r.RenterID) === normKey(rId)) return false;
        const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        return (cleanAadhDigits.length === 12 && rAadh === cleanAadhDigits) ||
               (prevAadhaarDigits.length === 12 && rAadh === prevAadhaarDigits);
      },
      function(r) {
        return {
          RenterName: record.RenterName || r.RenterName,
          IdentityDocNumber: record.IdentityDocNumber || r.IdentityDocNumber,
          AadhaarNumber: cleanAadhDigits || r.AadhaarNumber,
          MobileNumber: record.MobileNumber || r.MobileNumber,
          AltMobileNumber: record.AltMobileNumber || r.AltMobileNumber,
          Address: record.Address || r.Address,
          FamilyMembers: record.FamilyMembers || r.FamilyMembers,
          TotalFamilyMembers: record.TotalFamilyMembers || r.TotalFamilyMembers,
          ProfileImageURL: record.ProfileImageURL || r.ProfileImageURL || "",
          IdentityImageURL: record.IdentityImageURL || r.IdentityImageURL || "",
          IdentityBackImageURL: record.IdentityBackImageURL || r.IdentityBackImageURL || "",
          UpdatedAt: nowIso
        };
      }
    );

    // Sync updated Aadhaar number directly to any linked Registered User account in Users sheet
    if (cleanAadhDigits.length === 12) {
      const currentUsers = readSheetObjects("Users");
      const hasExplicitPrevMatch = prevAadhaarDigits.length === 12 && currentUsers.some(function(u) {
        const uA = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        return uA === prevAadhaarDigits;
      });
      updateRowsByMatcher(
        "Users",
        function(u) {
          const uAadh = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
          const matchByOldOrNew = (prevAadhaarDigits.length === 12 && uAadh === prevAadhaarDigits) || (uAadh === cleanAadhDigits);
          const matchByFlat = normKey(u.Role) === "renter" && record.PropertyName && record.FlatName &&
            normKey(u.AssignedProperty) === normKey(record.PropertyName) &&
            normKey(u.AssignedFlat) === normKey(record.FlatName) &&
            (!uAadh ||
             (prevAadhaarDigits.length === 12 && uAadh === prevAadhaarDigits) ||
             (!hasExplicitPrevMatch && normKey(record.Status || "Active") === "active" && normKey(u.Status || "Active") !== "blocked"));
          return matchByOldOrNew || matchByFlat;
        },
        function(u) {
          return {
            AssignedAadhaar: cleanAadhDigits,
            AadhaarNumber: cleanAadhDigits,
            FullName: record.RenterName || u.FullName || u.Username,
            UpdatedAt: nowIso
          };
        }
      );

      // Cascade old Aadhaar -> new Aadhaar across ALL existing Bills/Receipts, PaymentProofs, Monthly Photos, Screenshot Requests, Private Chats, Drive Metadata & GlobalConfig
      const matchedRenterIdSet = {};
      matchedRenterIdSet[normKey(rId)] = true;
      readSheetObjects("Renters").forEach(function(rr) {
        const rrA = String(rr.IdentityDocNumber || rr.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        if ((prevAadhaarDigits.length === 12 && rrA === prevAadhaarDigits) || rrA === cleanAadhDigits) {
          if (rr.RenterID) matchedRenterIdSet[normKey(rr.RenterID)] = true;
        }
      });

      const oldFormattedAadh = prevAadhaarDigits.length === 12
        ? (prevAadhaarDigits.slice(0, 4) + "-" + prevAadhaarDigits.slice(4, 8) + "-" + prevAadhaarDigits.slice(8, 12))
        : "";
      const newFormattedAadh = cleanAadhDigits.length === 12
        ? (cleanAadhDigits.slice(0, 4) + "-" + cleanAadhDigits.slice(4, 8) + "-" + cleanAadhDigits.slice(8, 12))
        : "";
      const replaceAadhInStr = function(val) {
        let s = String(val === undefined || val === null ? "" : val);
        if (!s || prevAadhaarDigits.length !== 12 || prevAadhaarDigits === cleanAadhDigits) return s;
        if (s.indexOf(prevAadhaarDigits) !== -1) s = s.split(prevAadhaarDigits).join(cleanAadhDigits);
        if (oldFormattedAadh && s.indexOf(oldFormattedAadh) !== -1) s = s.split(oldFormattedAadh).join(newFormattedAadh || cleanAadhDigits);
        return s;
      };

      updateRowsByMatcher(
        "Bills",
        function(b) {
          const bAadh = String(b.TenantAadhaar || b.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
          const bRid = normKey(b.RenterID);
          if (prevAadhaarDigits.length === 12 && bAadh === prevAadhaarDigits) return true;
          if (bRid && matchedRenterIdSet[bRid] && (!bAadh || (prevAadhaarDigits.length === 12 && bAadh === prevAadhaarDigits))) {
            return true;
          }
          if (!bAadh && record.PropertyName && record.FlatName &&
              normKey(b.PropertyName) === normKey(record.PropertyName) &&
              normKey(b.FlatName) === normKey(record.FlatName) &&
              (!record.RenterName || !b.TenantName || normKey(b.TenantName) === normKey(record.RenterName) || (prevRow.RenterName && normKey(b.TenantName) === normKey(prevRow.RenterName)))) {
            return true;
          }
          return false;
        },
        function() {
          return {
            TenantAadhaar: cleanAadhDigits,
            UpdatedAt: nowIso
          };
        }
      );

      if (prevAadhaarDigits.length === 12 && prevAadhaarDigits !== cleanAadhDigits) {
        updateRowsByMatcher(
          "PaymentProofs",
          function(p) {
            const pAadh = String(p.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
            const pRid = normKey(p.RenterID || "");
            return pAadh === prevAadhaarDigits || Boolean(pRid && matchedRenterIdSet[pRid] && (!pAadh || pAadh === prevAadhaarDigits));
          },
          function() {
            return {
              AadhaarNumber: cleanAadhDigits,
              UpdatedAt: nowIso
            };
          }
        );

        try {
          updateRowsByMatcher(
            "PrivateChats",
            function(c) {
              const ch = String(c.ChannelKey || "");
              return ch.indexOf(prevAadhaarDigits) !== -1 || (oldFormattedAadh && ch.indexOf(oldFormattedAadh) !== -1);
            },
            function(c) {
              return {
                ChannelKey: replaceAadhInStr(c.ChannelKey || "")
              };
            }
          );
        } catch (chatSheetErr) {}

        try {
          const cfgRows = readSheetObjects("GlobalConfig");
          const getCfgParsed = function(keyLower, fallbackVal) {
            const row = cfgRows.find(function(r) { return normKey(r.ConfigKey) === keyLower; });
            if (!row || !row.ConfigValueJSON) return fallbackVal;
            try { return JSON.parse(row.ConfigValueJSON); } catch (e) { return fallbackVal; }
          };

          const ppList = getCfgParsed("paymentproofs", null);
          if (Array.isArray(ppList)) {
            let ppChanged = false;
            const nextPpList = ppList.map(function(p) {
              if (!p) return p;
              const pa = String(p.aadhaar || p.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
              const prid = normKey(p.renterId || p.RenterID || "");
              if (pa === prevAadhaarDigits || (prid && matchedRenterIdSet[prid] && (!pa || pa === prevAadhaarDigits))) {
                ppChanged = true;
                return Object.assign({}, p, { aadhaar: cleanAadhDigits, AadhaarNumber: cleanAadhDigits });
              }
              return p;
            });
            if (ppChanged) handleSaveConfigItem("paymentProofs", nextPpList);
          }

          const rmpMap = getCfgParsed("rentermonthlyphotos", null);
          if (rmpMap && typeof rmpMap === "object" && !Array.isArray(rmpMap)) {
            let rmpChanged = false;
            const nextRmpMap = {};
            Object.keys(rmpMap).forEach(function(k) {
              const slot = rmpMap[k];
              if (!slot || typeof slot !== "object") {
                nextRmpMap[k] = slot;
                return;
              }
              const sa = String(slot.aadhaar || slot.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
              const srid = normKey(slot.renterId || "");
              const keyHasOld = k.indexOf(prevAadhaarDigits) !== -1 || (oldFormattedAadh && k.indexOf(oldFormattedAadh) !== -1);
              const slotMatches = sa === prevAadhaarDigits || keyHasOld || (srid && matchedRenterIdSet[srid] && (!sa || sa === prevAadhaarDigits));
              const newKey = keyHasOld ? replaceAadhInStr(k) : k;
              if (slotMatches || newKey !== k) {
                rmpChanged = true;
                const updatedSlot = Object.assign({}, slot, {
                  aadhaar: cleanAadhDigits,
                  identityDocNumber: cleanAadhDigits,
                  slotKey: slot.slotKey ? replaceAadhInStr(slot.slotKey) : newKey
                });
                nextRmpMap[newKey] = updatedSlot;
              } else {
                nextRmpMap[k] = slot;
              }
            });
            if (rmpChanged) handleSaveConfigItem("renterMonthlyPhotos", nextRmpMap);
          }

          const srMap = getCfgParsed("screenshotrequests", null);
          if (srMap && typeof srMap === "object" && !Array.isArray(srMap)) {
            let srChanged = false;
            const nextSrMap = {};
            Object.keys(srMap).forEach(function(k) {
              const entry = srMap[k];
              const keyHasOld = k.indexOf(prevAadhaarDigits) !== -1 || (oldFormattedAadh && k.indexOf(oldFormattedAadh) !== -1);
              const newKey = keyHasOld ? replaceAadhInStr(k) : k;
              if (entry && typeof entry === "object") {
                const ea = String(entry.aadhaar || entry.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
                const erid = normKey(entry.renterId || "");
                const entryMatches = ea === prevAadhaarDigits || keyHasOld || (erid && matchedRenterIdSet[erid] && (!ea || ea === prevAadhaarDigits));
                if (entryMatches || newKey !== k) {
                  srChanged = true;
                  nextSrMap[newKey] = Object.assign({}, entry, {
                    aadhaar: cleanAadhDigits,
                    requestKey: entry.requestKey ? replaceAadhInStr(entry.requestKey) : newKey
                  });
                } else {
                  nextSrMap[k] = entry;
                }
              } else {
                if (newKey !== k) srChanged = true;
                nextSrMap[newKey] = entry;
              }
            });
            if (srChanged) handleSaveConfigItem("screenshotRequests", nextSrMap);
          }

          const urList = getCfgParsed("userrequests", null);
          if (Array.isArray(urList)) {
            let urChanged = false;
            const nextUr = urList.map(function(req) {
              if (!req || typeof req !== "object") return req;
              const ra = String(req.aadhaar || req.AadhaarNumber || req.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
              if (ra === prevAadhaarDigits) {
                urChanged = true;
                return Object.assign({}, req, {
                  aadhaar: cleanAadhDigits,
                  AadhaarNumber: cleanAadhDigits,
                  identityDocNumber: req.identityDocNumber ? cleanAadhDigits : req.identityDocNumber
                });
              }
              return req;
            });
            if (urChanged) handleSaveConfigItem("userRequests", nextUr);
          }

          const mtList = getCfgParsed("maintenancetickets", null);
          if (Array.isArray(mtList)) {
            let mtChanged = false;
            const nextMt = mtList.map(function(t) {
              if (!t || typeof t !== "object") return t;
              const ta = String(t.aadhaar || t.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
              if (ta === prevAadhaarDigits) {
                mtChanged = true;
                return Object.assign({}, t, { aadhaar: cleanAadhDigits });
              }
              return t;
            });
            if (mtChanged) handleSaveConfigItem("maintenanceTickets", nextMt);
          }

          const pcMap = getCfgParsed("privatechats", null);
          if (pcMap && typeof pcMap === "object" && !Array.isArray(pcMap)) {
            let pcChanged = false;
            const nextPc = {};
            Object.keys(pcMap).forEach(function(k) {
              const newKey = replaceAadhInStr(k);
              if (newKey !== k) pcChanged = true;
              const msgs = Array.isArray(pcMap[k]) ? pcMap[k].map(function(m) {
                if (!m || typeof m !== "object") return m;
                const ma = String(m.aadhaar || "").replace(/\D/g, "").slice(0, 12);
                if (ma === prevAadhaarDigits) {
                  pcChanged = true;
                  return Object.assign({}, m, { aadhaar: cleanAadhDigits });
                }
                return m;
              }) : pcMap[k];
              if (Array.isArray(nextPc[newKey]) && Array.isArray(msgs)) {
                nextPc[newKey] = nextPc[newKey].concat(msgs);
              } else {
                nextPc[newKey] = msgs;
              }
            });
            if (pcChanged) handleSaveConfigItem("privateChats", nextPc);
          }
        } catch (cfgErr) {}

        // Update Google Drive photo file descriptions so hydratePhotosFromDriveFolder never re-hydrates old Aadhaar
        try {
          const folder = getPortalDrivePhotoFolder();
          const files = folder.getFiles();
          while (files.hasNext()) {
            const f = files.next();
            const desc = String(f.getDescription() || "").trim();
            if (!desc || desc.charAt(0) !== "{") continue;
            let meta = null;
            try { meta = JSON.parse(desc); } catch (e) { meta = null; }
            if (!meta || typeof meta !== "object") continue;
            const mAadh = String(meta.aadhaar || meta.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
            const mSlot = String(meta.slotKey || "");
            const mRid = normKey(meta.renterId || "");
            const matchesOld = (mAadh === prevAadhaarDigits) ||
              (mSlot && (mSlot.indexOf(prevAadhaarDigits) !== -1 || (oldFormattedAadh && mSlot.indexOf(oldFormattedAadh) !== -1))) ||
              (mRid && matchedRenterIdSet[mRid] && (!mAadh || mAadh === prevAadhaarDigits));
            if (matchesOld) {
              meta.aadhaar = cleanAadhDigits;
              meta.identityDocNumber = cleanAadhDigits;
              if (meta.slotKey) meta.slotKey = replaceAadhInStr(meta.slotKey);
              f.setDescription(JSON.stringify(meta));
            }
          }
        } catch (driveMetaErr) {}
      }
    }
  }

  // Auto-block linked Renter User ID when renter vacates flat (Status becomes Inactive) and has no other Active stay
  if (normKey(statusVal) === "inactive") {
    const refreshedRenters = readSheetObjects("Renters");
    const hasOtherActiveStay = refreshedRenters.some(function(r) {
      if (normKey(r.Status) !== "active") return false;
      const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
      return (cleanAadhDigits.length === 12 && rAadh === cleanAadhDigits) ||
             (prevAadhaarDigits.length === 12 && rAadh === prevAadhaarDigits);
    });
    const flatHasNewActiveRenter = record.PropertyName && record.FlatName && refreshedRenters.some(function(r) {
      return normKey(r.PropertyName) === normKey(record.PropertyName) &&
             normKey(r.FlatName) === normKey(record.FlatName) &&
             normKey(r.Status) === "active";
    });

    if (!hasOtherActiveStay) {
      updateRowsByMatcher(
        "Users",
        function(u) {
          if (normKey(u.Role) !== "renter") return false;
          const uAadh = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
          const matchByAadhaar = (cleanAadhDigits.length === 12 && uAadh === cleanAadhDigits) ||
                                 (prevAadhaarDigits.length === 12 && uAadh === prevAadhaarDigits);
          const matchByFlat = !flatHasNewActiveRenter && record.PropertyName && record.FlatName &&
            normKey(u.AssignedProperty) === normKey(record.PropertyName) &&
            normKey(u.AssignedFlat) === normKey(record.FlatName) &&
            (!uAadh || (cleanAadhDigits.length === 12 && uAadh === cleanAadhDigits) || (prevAadhaarDigits.length === 12 && uAadh === prevAadhaarDigits));
          return matchByAadhaar || matchByFlat;
        },
        function() {
          return {
            Status: "Blocked",
            AccountStatus: "Blocked",
            MovedOutAutoBlocked: "true",
            ActiveSessionToken: "",
            UpdatedAt: nowIso
          };
        }
      );
    }
  } else if (normKey(statusVal) === "active") {
    // Auto-unblock linked blocked Renter User ID when renter becomes Active again (Add Renter / Reactivate / Shift / Profile Save)
    const unblockedUsernames = [];
    updateRowsByMatcher(
      "Users",
      function(u) {
        if (normKey(u.Role) !== "renter") return false;
        const uStatus = normKey(u.Status || u.AccountStatus || "active");
        const isBlockedUser = uStatus === "blocked";
        if (!isBlockedUser) return false;
        const uAadh = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        const matchByAadhaar = (cleanAadhDigits.length === 12 && uAadh === cleanAadhDigits) ||
                               (prevAadhaarDigits.length === 12 && uAadh === prevAadhaarDigits);
        const wasMovedOutBlocked = String(u.MovedOutAutoBlocked || "").toLowerCase() === "true";
        const matchByMovedOutFlat = wasMovedOutBlocked && !uAadh && record.PropertyName && record.FlatName &&
          normKey(u.AssignedProperty) === normKey(record.PropertyName) &&
          normKey(u.AssignedFlat) === normKey(record.FlatName);
        if (matchByAadhaar || matchByMovedOutFlat) {
          unblockedUsernames.push(normKey(u.Username));
          return true;
        }
        return false;
      },
      function(u) {
        const nextUpdates = {
          Status: "Active",
          AccountStatus: "Active",
          MovedOutAutoBlocked: "false",
          LockoutStage: 0,
          FailedAttempts: 0,
          LockedUntil: 0,
          LockedTierLabel: "",
          IsPermanentlyBlocked: "false",
          UpdatedAt: nowIso
        };
        if (record.PropertyName) nextUpdates.AssignedProperty = record.PropertyName;
        if (record.FlatName) nextUpdates.AssignedFlat = record.FlatName;
        if (cleanAadhDigits.length === 12) {
          nextUpdates.AssignedAadhaar = cleanAadhDigits;
          nextUpdates.AadhaarNumber = cleanAadhDigits;
        }
        if (record.RenterName) nextUpdates.FullName = record.RenterName;
        return nextUpdates;
      }
    );
    if (unblockedUsernames.length > 0) {
      try {
        const cfgRows = readSheetObjects("GlobalConfig");
        const lockRow = cfgRows.find(function(r) { return normKey(r.ConfigKey) === "loginlockouts"; });
        if (lockRow && lockRow.ConfigValueJSON) {
          const parsedLocks = JSON.parse(lockRow.ConfigValueJSON) || {};
          unblockedUsernames.forEach(function(unameLower) { delete parsedLocks[unameLower]; });
          handleSaveConfigItem("loginLockouts", parsedLocks);
        }
      } catch (e) {}
    }
  }

  const refreshedBills = readSheetObjects("Bills").map(function(b) {
    b.MonthYear = String(b.MonthYear || "").replace(/^'+/, "");
    b.BillDate = String(b.BillDate || "").replace(/^'+/, "");
    b.PaymentDate = String(b.PaymentDate || "").replace(/^'+/, "");
    return b;
  });

  const refreshedRentersOut = readSheetObjects("Renters").map(function(r) {
    if (typeof r.FamilyMembersList === "string" && r.FamilyMembersList.trim()) {
      try { r.FamilyMembersList = JSON.parse(r.FamilyMembersList); } catch (e) { r.FamilyMembersList = []; }
    } else if (!Array.isArray(r.FamilyMembersList)) {
      r.FamilyMembersList = [];
    }
    return r;
  });

  const refreshedProofsOut = readSheetObjects("PaymentProofs")
    .filter(function(p) { return Boolean(p.ProofID || p.UTRNumber || p.AmountPaid || p.ScreenshotURL); })
    .map(function(p) {
      const scrUrl = String(p.ScreenshotURL || "").trim();
      return {
        id: p.ProofID || ("PRF-" + Date.now()),
        billId: p.BillID || "",
        propertyName: p.PropertyName || "",
        flatName: p.FlatName || "",
        monthYear: String(p.MonthYear || "").replace(/^'+/, ""),
        renterName: p.RenterName || "",
        username: p.Username || "",
        aadhaar: String(p.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12),
        amount: Number(p.AmountPaid || 0),
        utr: p.UTRNumber || "",
        paymentMode: p.PaymentMode || "UPI",
        status: p.Status || "Pending",
        submittedAt: p.SubmittedAt || "",
        verifiedBy: p.VerifiedBy || "",
        verifiedAt: p.VerifiedAt || "",
        notes: p.Notes || "",
        rejectionReason: String(p.Status || "").toLowerCase() === "rejected" ? (p.Notes || "") : "",
        screenshotUrl: scrUrl,
        imageUrl: scrUrl,
        proofImageUrl: scrUrl
      };
    });

  let refreshedMonthlyPhotosOut = null;
  let refreshedScreenshotRequestsOut = null;
  try {
    const outCfgRows = readSheetObjects("GlobalConfig");
    const rmpOutRow = outCfgRows.find(function(r) { return normKey(r.ConfigKey) === "rentermonthlyphotos"; });
    if (rmpOutRow && rmpOutRow.ConfigValueJSON) refreshedMonthlyPhotosOut = JSON.parse(rmpOutRow.ConfigValueJSON);
    const srOutRow = outCfgRows.find(function(r) { return normKey(r.ConfigKey) === "screenshotrequests"; });
    if (srOutRow && srOutRow.ConfigValueJSON) refreshedScreenshotRequestsOut = JSON.parse(srOutRow.ConfigValueJSON);
  } catch (e) {}

  return {
    status: "success",
    renterId: rId,
    profileImageUrl: profilePhotoLink,
    identityImageUrl: identityPhotoLink,
    identityBackImageUrl: identityBackPhotoLink,
    renters: refreshedRentersOut,
    users: readSheetObjects("Users").filter(function(u) { return Boolean(u.Username); }),
    bills: refreshedBills,
    paymentProofs: refreshedProofsOut,
    renterMonthlyPhotos: refreshedMonthlyPhotosOut,
    screenshotRequests: refreshedScreenshotRequestsOut,
    message: "Renter profile and flat rate saved to Google Sheet and Google Drive."
  };
}

function handleUploadRenterDocument(data) {
  const pName = String(data.propertyName || "").trim();
  const fName = String(data.flatName || "").trim();
  const rId = String(data.renterId || "").trim();
  const rName = String(data.renterName || "Renter").trim();
  const aadhaar = String(data.identityDocNumber || data.aadhaar || "").replace(/\D/g, "").slice(0, 12);
  const docType = String(data.docType || "identity").trim().toLowerCase();
  const isIdentityBack = docType === "identity_back" || docType === "aadhaar_back";
  const isIdentity = docType === "identity" || docType === "identity_front";
  const rawPhotoData = String(data.base64Data || data.sheetSafeBase64 || data.directImageUrl || "").trim();
  const nowIso = new Date().toISOString();

  const resolvedRenterId = rId || ("R-" + Date.now());
  const slotKey = "RENTER_DOC__" + sanitizeDriveFilePart(resolvedRenterId || (pName + "_" + fName)) + "__" + docType;
  const docExt = (String(data.mimeType || "").toLowerCase().indexOf("png") !== -1 || rawPhotoData.indexOf("data:image/png") === 0) ? ".png" : ".jpg";
  const fileName = "Renter_" + sanitizeDriveFilePart(docType) + "_" +
    sanitizeDriveFilePart(pName) + "_Flat_" + sanitizeDriveFilePart(fName) + "_" +
    sanitizeDriveFilePart(rName) + docExt;

  const driveSaved = savePhotoToDriveFolderOnly(rawPhotoData, fileName, {
    kind: "RENTER_DOCUMENT",
    slotKey: slotKey,
    docType: docType,
    renterId: resolvedRenterId,
    propertyName: pName,
    flatName: fName,
    renterName: rName,
    aadhaar: aadhaar,
    identityDocNumber: aadhaar
  });

  const driveShareLink = driveSaved.shareableUrl || driveSaved.previewUrl;
  const photoPatch = { UpdatedAt: nowIso };
  if (isIdentityBack) {
    photoPatch.IdentityBackImageURL = driveShareLink;
  } else if (isIdentity) {
    photoPatch.IdentityImageURL = driveShareLink;
  } else if (docType === "photo1" || docType === "photo_1") {
    photoPatch.Photo1URL = driveShareLink;
  } else if (docType === "photo2" || docType === "photo_2") {
    photoPatch.Photo2URL = driveShareLink;
  } else if (docType === "photo3" || docType === "photo_3") {
    photoPatch.Photo3URL = driveShareLink;
  } else if (docType === "photo4" || docType === "photo_4") {
    photoPatch.Photo4URL = driveShareLink;
  } else {
    photoPatch.ProfileImageURL = driveShareLink;
  }

  const existingRenters = readSheetObjects("Renters");
  const matchedExisting = existingRenters.some(function(r) {
    if (rId && normKey(r.RenterID) === normKey(rId)) return true;
    const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
    if (aadhaar.length === 12 && rAadh === aadhaar) return true;
    return Boolean(pName && fName && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName) && normKey(r.Status || "Active") === "active");
  });

  if (matchedExisting) {
    // Update all matching renter rows (by RenterID, 12-digit Aadhaar, or Active flat record) with the Google Drive shareable link
    updateRowsByMatcher(
      "Renters",
      function(r) {
        if (rId && normKey(r.RenterID) === normKey(rId)) return true;
        const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        if (aadhaar.length === 12 && rAadh === aadhaar) return true;
        return Boolean(pName && fName && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName) && normKey(r.Status || "Active") === "active");
      },
      function() {
        return Object.assign({}, photoPatch);
      }
    );
  } else {
    // If photo was uploaded before clicking 'Save Renter Record', create the Renters row immediately so the Drive link is stored in the DB
    const newRecord = Object.assign({
      RenterID: resolvedRenterId,
      PropertyName: pName,
      FlatName: fName,
      RenterName: rName,
      RentAmount: Number(data.rentAmount) || 0,
      JoinDate: String(data.joinDate || nowIso.slice(0, 10)).trim(),
      EndDate: "",
      Status: "Active",
      IdentityDocNumber: aadhaar,
      AadhaarNumber: aadhaar
    }, photoPatch);
    upsertRowByMatcher(
      "Renters",
      function(r) { return normKey(r.RenterID) === normKey(resolvedRenterId); },
      newRecord
    );
  }

  return {
    status: "success",
    resolvedRenterId: resolvedRenterId,
    driveFileId: driveSaved.fileId,
    fileId: driveSaved.fileId,
    fileName: driveSaved.fileName,
    driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
    driveFolderUrl: GOOGLE_DRIVE_PHOTO_FOLDER_URL,
    shareableUrl: driveShareLink,
    driveUrl: driveSaved.driveUrl || driveShareLink,
    imageUrl: driveShareLink,
    previewUrl: driveSaved.previewUrl,
    message: "Photo saved to Google Drive and its shareable link stored in the database."
  };
}

function handleUploadRenterMonthlyPhoto(data) {
  const pName = String(data.propertyName || "Main Property").trim();
  const fName = String(data.flatName || "--").trim();
  const rId = String(data.renterId || "").trim();
  const rName = String(data.renterName || "Renter").trim();
  const uName = String(data.username || "").trim();
  let aadhaar = String(data.identityDocNumber || data.aadhaar || "").replace(/\D/g, "").slice(0, 12);
  const monthYear = String(data.monthYear || new Date().toISOString().slice(0, 7)).replace(/^'+/, "").trim();
  const category = canonicalMonthlyCategory(data.category || "electricity_meter") || "electricity_meter";
  const categoryLabel = String(data.categoryLabel || category).trim();
  const rawPhotoData = String(data.base64Data || data.sheetSafeBase64 || "").trim();
  const nowIso = new Date().toISOString();

  if (!rawPhotoData) {
    return { status: "error", message: "Photo data is required." };
  }

  // Fallback resolve 12-digit Aadhaar from Renters or Users sheets when caller didn't send it
  if (aadhaar.length !== 12) {
    try {
      const allRenters = readSheetObjects("Renters");
      const matchedR = allRenters.find(function(r) {
        if (rId && normKey(r.RenterID) === normKey(rId)) return true;
        return Boolean(pName && fName && fName !== "--" && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName) && normKey(r.Status || "Active") === "active");
      });
      if (matchedR) {
        aadhaar = String(matchedR.IdentityDocNumber || matchedR.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
      }
      if (aadhaar.length !== 12) {
        const allUsers = readSheetObjects("Users");
        const matchedU = allUsers.find(function(u) {
          if (normKey(u.Role) !== "renter") return false;
          if (uName && normKey(u.Username) === normKey(uName)) return true;
          return Boolean(pName && fName && fName !== "--" && normKey(u.AssignedProperty) === normKey(pName) && normKey(u.AssignedFlat) === normKey(fName));
        });
        if (matchedU) {
          aadhaar = String(matchedU.AssignedAadhaar || matchedU.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        }
      }
    } catch (lookupErr) {}
  }

  const slotKey = (aadhaar.length === 12)
    ? ("MONTHLY_AADHAAR__" + aadhaar + "__" + normKey(monthYear) + "__" + normKey(category))
    : ((fName && fName !== "--")
      ? ("MONTHLY__" + normKey(pName) + "__" + normKey(fName) + "__" + normKey(monthYear) + "__" + normKey(category))
      : ("MONTHLY_USER__" + normKey(uName || rName) + "__" + normKey(monthYear) + "__" + normKey(category)));
  const monthlyExt = (String(data.mimeType || "").toLowerCase().indexOf("png") !== -1 || rawPhotoData.indexOf("data:image/png") === 0) ? ".png" : ".jpg";
  const fileName = "Monthly_" + sanitizeDriveFilePart(monthYear) + "_" +
    sanitizeDriveFilePart(category) + "_" +
    sanitizeDriveFilePart(aadhaar || (pName + "_Flat_" + fName)) + "_" +
    sanitizeDriveFilePart(rName) + monthlyExt;

  let driveSaved;
  try {
    driveSaved = savePhotoToDriveFolderOnly(rawPhotoData, fileName, {
      kind: "MONTHLY_UTILITY_PHOTO",
      slotKey: slotKey,
      propertyName: pName,
      flatName: fName,
      monthYear: monthYear,
      category: category,
      categoryLabel: categoryLabel,
      renterId: rId,
      renterName: rName,
      username: uName,
      aadhaar: aadhaar,
      identityDocNumber: aadhaar
    });
  } catch (driveErr) {
    return {
      status: "error",
      message: "Google Drive upload failed: " + (driveErr && driveErr.message ? driveErr.message : driveErr)
    };
  }

  const photoRecord = {
    slotKey: slotKey,
    driveFileId: driveSaved.fileId,
    fileId: driveSaved.fileId,
    fileName: driveSaved.fileName,
    imageUrl: driveSaved.previewUrl,
    previewUrl: driveSaved.previewUrl,
    shareableUrl: driveSaved.shareableUrl,
    driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
    driveFolderUrl: GOOGLE_DRIVE_PHOTO_FOLDER_URL,
    propertyName: pName,
    flatName: fName,
    monthYear: monthYear,
    category: category,
    categoryLabel: categoryLabel,
    renterId: rId,
    renterName: rName,
    username: uName,
    aadhaar: aadhaar,
    identityDocNumber: aadhaar,
    uploadedAt: driveSaved.uploadedAt
  };

  // Paste the Google Drive link into GlobalConfig.renterMonthlyPhotos AND into the active Renters sheet Photo1URL..Photo4URL columns!
  let colTarget = "";
  if (category === "electricity_meter") colTarget = "Photo1URL";
  else if (category === "completed_payment") colTarget = "Photo2URL";
  else if (category === "water_meter_bill") colTarget = "Photo3URL";
  else if (category === "png_gas_meter") colTarget = "Photo4URL";

  if (colTarget) {
    updateRowsByMatcher(
      "Renters",
      function(r) {
        if (rId && normKey(r.RenterID) === normKey(rId)) return true;
        const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        if (aadhaar.length === 12 && rAadh === aadhaar && normKey(r.Status || "Active") === "active") return true;
        if (aadhaar.length === 12 && rAadh === aadhaar && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName)) return true;
        return Boolean(pName && fName && fName !== "--" && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName) && normKey(r.Status || "Active") === "active");
      },
      function() {
        const patch = { UpdatedAt: nowIso };
        patch[colTarget] = driveSaved.previewUrl;
        return patch;
      }
    );
  }

  // Persist in GlobalConfig.renterMonthlyPhotos (kept permanently unless Admin manually deletes)
  const rawCfgRows = readSheetObjects("GlobalConfig");
  let existingMap = {};
  const foundCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "rentermonthlyphotos"; });
  if (foundCfg && foundCfg.ConfigValueJSON) {
    try { existingMap = JSON.parse(foundCfg.ConfigValueJSON) || {}; } catch (e) { existingMap = {}; }
  }
  const legacyMonthlyKey = (fName && fName !== "--") ? ("MONTHLY__" + normKey(pName) + "__" + normKey(fName) + "__" + normKey(monthYear) + "__" + normKey(category)) : "";
  const flatSlotKey = (fName && fName !== "--") ? (normKey(pName) + "__" + normKey(fName) + "__" + normKey(monthYear) + "__" + normKey(category)) : "";
  const aadhaarSlotKey = aadhaar.length === 12 ? (aadhaar + "__" + normKey(monthYear) + "__" + normKey(category)) : "";
  if (legacyMonthlyKey && legacyMonthlyKey !== slotKey) delete existingMap[legacyMonthlyKey];
  if (flatSlotKey && flatSlotKey !== slotKey) delete existingMap[flatSlotKey];
  if (aadhaarSlotKey && aadhaarSlotKey !== slotKey) delete existingMap[aadhaarSlotKey];
  existingMap[slotKey] = photoRecord;

  const prunedKeys = [];
  try {
    // Always preserve at least 3+ months (90+ days) of uploaded photo history; only prune > 90-day entries if JSON nears Sheet cell cap
    if (JSON.stringify(existingMap).length > 48000) {
      const cutoffMs = Date.now() - (95 * 24 * 60 * 60 * 1000);
      const oldKeys = Object.keys(existingMap).filter(function(k) {
        if (k === slotKey) return false;
        const entry = existingMap[k];
        if (!entry || typeof entry !== "object") return true;
        const ts = Date.parse(entry.updatedAt || entry.uploadedAt || "") || 0;
        return ts > 0 && ts < cutoffMs;
      }).sort(function(a, b) {
        const ta = Date.parse(existingMap[a].updatedAt || existingMap[a].uploadedAt || "") || 0;
        const tb = Date.parse(existingMap[b].updatedAt || existingMap[b].uploadedAt || "") || 0;
        return ta - tb;
      });
      for (let i = 0; i < oldKeys.length && JSON.stringify(existingMap).length > 47500; i++) {
        delete existingMap[oldKeys[i]];
        prunedKeys.push(oldKeys[i]);
      }
    }
    if (JSON.stringify(existingMap).length <= 49500) {
      handleSaveConfigItem("renterMonthlyPhotos", existingMap);
    }
  } catch (cfgSaveErr) {}

  // Mark matching targeted/re-request screenshotRequests entries as submitted
  try {
    const foundReqCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "screenshotrequests"; });
    if (foundReqCfg && foundReqCfg.ConfigValueJSON) {
      const reqMap = JSON.parse(foundReqCfg.ConfigValueJSON) || {};
      let reqChanged = false;
      Object.keys(reqMap).forEach(function(k) {
        const entry = reqMap[k];
        if (!entry || typeof entry !== "object") return;
        const eMonth = String(entry.monthYear || "").replace(/^'+/, "").trim();
        if (normKey(eMonth) !== normKey(monthYear)) return;
        const eCat = canonicalMonthlyCategory(entry.category || "");
        if (eCat && eCat !== category) return;
        const scope = String(entry.scope || "").toUpperCase();
        if (scope === "ALL_ACTIVE") return;
        const eAadh = String(entry.aadhaar || entry.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
        const matchRenter = (rId && entry.renterId && normKey(entry.renterId) === normKey(rId)) ||
          (aadhaar.length === 12 && eAadh === aadhaar) ||
          (pName && fName && fName !== "--" && normKey(entry.propertyName) === normKey(pName) && normKey(entry.flatName) === normKey(fName));
        if (matchRenter) {
          entry.status = "submitted";
          entry.submittedAt = nowIso;
          reqChanged = true;
        }
      });
      if (reqChanged) {
        handleSaveConfigItem("screenshotRequests", reqMap);
      }
    }
  } catch (e) {}

  return {
    status: "success",
    slotKey: slotKey,
    driveFileId: driveSaved.fileId,
    fileId: driveSaved.fileId,
    fileName: driveSaved.fileName,
    shareableUrl: driveSaved.shareableUrl,
    imageUrl: driveSaved.previewUrl,
    previewUrl: driveSaved.previewUrl,
    driveFolderId: GOOGLE_DRIVE_PHOTO_FOLDER_ID,
    driveFolderUrl: GOOGLE_DRIVE_PHOTO_FOLDER_URL,
    uploadedAt: driveSaved.uploadedAt,
    photoRecord: photoRecord,
    prunedSlotKeys: prunedKeys,
    renterMonthlyPhotos: existingMap,
    message: categoryLabel + " saved permanently to Google Drive."
  };
}

function handleDeleteRenterMonthlyPhoto(data) {
  if (data.requesterRole && normKey(data.requesterRole) !== "admin") {
    return {
      status: "error",
      message: "Permission Denied: Only Admin can delete uploaded photos."
    };
  }
  const pName = String(data.propertyName || "Main Property").trim();
  const fName = String(data.flatName || "--").trim();
  const rId = String(data.renterId || "").trim();
  const aadhaar = String(data.identityDocNumber || data.aadhaar || "").replace(/\D/g, "").slice(0, 12);
  const monthYear = String(data.monthYear || "").replace(/^'+/, "").trim();
  const category = String(data.category || "").trim().toLowerCase();
  const slotKey = String(data.slotKey || ("MONTHLY__" + normKey(pName) + "__" + normKey(fName) + "__" + normKey(monthYear) + "__" + normKey(category))).trim();
  const altFlatSlotKey = normKey(pName) + "__" + normKey(fName) + "__" + normKey(monthYear) + "__" + normKey(category);
  const aadhaarSlotKey = aadhaar.length === 12 ? (aadhaar + "__" + normKey(monthYear) + "__" + normKey(category)) : "";
  const altAadhaarMonthlyKey = aadhaar.length === 12 ? ("MONTHLY_AADHAAR__" + aadhaar + "__" + normKey(monthYear) + "__" + normKey(category)) : "";
  const nowIso = new Date().toISOString();

  deletePhotoFromDriveFolderBySlot(slotKey);
  if (altFlatSlotKey && altFlatSlotKey !== slotKey) {
    deletePhotoFromDriveFolderBySlot(altFlatSlotKey);
  }
  if (altAadhaarMonthlyKey && altAadhaarMonthlyKey !== slotKey) {
    deletePhotoFromDriveFolderBySlot(altAadhaarMonthlyKey);
  }
  if (data.fileId) {
    try {
      DriveApp.getFileById(String(data.fileId)).setTrashed(true);
    } catch (e) {}
  }

  // Remove from GlobalConfig.renterMonthlyPhotos (including all matching alias keys for that Aadhaar/flat + month + category or fileId)
  const canonCatDel = canonicalMonthlyCategory(category);
  const rawCfgRows = readSheetObjects("GlobalConfig");
  const foundCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "rentermonthlyphotos"; });
  let updatedMap = null;
  if (foundCfg && foundCfg.ConfigValueJSON) {
    try {
      const existingMap = JSON.parse(foundCfg.ConfigValueJSON) || {};
      delete existingMap[slotKey];
      if (altFlatSlotKey) delete existingMap[altFlatSlotKey];
      if (aadhaarSlotKey) delete existingMap[aadhaarSlotKey];
      if (altAadhaarMonthlyKey) delete existingMap[altAadhaarMonthlyKey];
      Object.keys(existingMap).forEach(function(k) {
        const entry = existingMap[k];
        if (!entry || typeof entry !== "object") return;
        const eFileId = String(entry.driveFileId || entry.fileId || "").trim();
        if (data.fileId && eFileId === String(data.fileId).trim()) {
          if (entry.slotKey) deletePhotoFromDriveFolderBySlot(entry.slotKey);
          delete existingMap[k];
          return;
        }
        const eMonth = String(entry.monthYear || entry.MonthYear || "").replace(/^'+/, "").trim();
        const eCat = canonicalMonthlyCategory(entry.category || "");
        if (monthYear && eMonth === monthYear && canonCatDel && eCat === canonCatDel) {
          const eAadh = String(entry.aadhaar || entry.identityDocNumber || "").replace(/\D/g, "").slice(0, 12);
          const sameA = aadhaar.length === 12 && (eAadh === aadhaar || k.indexOf(aadhaar) !== -1);
          const sameId = rId && entry.renterId && normKey(entry.renterId) === normKey(rId);
          const sameFlat = pName && fName && fName !== "--" && normKey(entry.propertyName) === normKey(pName) && normKey(entry.flatName) === normKey(fName);
          if (sameA || sameId || sameFlat) {
            if (entry.slotKey) deletePhotoFromDriveFolderBySlot(entry.slotKey);
            delete existingMap[k];
          }
        }
      });
      updatedMap = existingMap;
      handleSaveConfigItem("renterMonthlyPhotos", existingMap);
    } catch (e) {}
  }

  // Clear corresponding Photo1URL..Photo4URL on matching Renter row
  let colTarget = "";
  if (canonCatDel === "electricity_meter") colTarget = "Photo1URL";
  else if (canonCatDel === "completed_payment") colTarget = "Photo2URL";
  else if (canonCatDel === "water_meter_bill") colTarget = "Photo3URL";
  else if (canonCatDel === "png_gas_meter") colTarget = "Photo4URL";

  if (colTarget) {
    updateRowsByMatcher(
      "Renters",
      function(r) {
        if (rId && normKey(r.RenterID) === normKey(rId)) return true;
        const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
        if (aadhaar.length === 12 && rAadh === aadhaar && normKey(r.Status || "Active") === "active") return true;
        if (aadhaar.length === 12 && rAadh === aadhaar && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName)) return true;
        return Boolean(pName && fName && fName !== "--" && normKey(r.PropertyName) === normKey(pName) && normKey(r.FlatName) === normKey(fName) && normKey(r.Status || "Active") === "active");
      },
      function() {
        const patch = { UpdatedAt: nowIso };
        patch[colTarget] = "";
        return patch;
      }
    );
  }

  return {
    status: "success",
    slotKey: slotKey,
    renterMonthlyPhotos: updatedMap,
    message: "Monthly photo deleted by Admin from Google Drive and Google Sheets."
  };
}

function handleSaveScreenshotRequests(data) {
  const nowIso = new Date().toISOString();
  const rawCfgRows = readSheetObjects("GlobalConfig");
  let existingRequests = {};
  const foundCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "screenshotrequests"; });
  if (foundCfg && foundCfg.ConfigValueJSON) {
    try { existingRequests = JSON.parse(foundCfg.ConfigValueJSON) || {}; } catch (e) { existingRequests = {}; }
  }

  const targetMonth = String(data.monthYear || (data.requestEntry && data.requestEntry.monthYear) || "").replace(/^'+/, "").trim();

  if (data.action === "closeMonthlyPhotoRequest" && targetMonth) {
    const allActiveKey = "ALL_ACTIVE__" + targetMonth;
    existingRequests[allActiveKey] = {
      requestKey: allActiveKey,
      scope: "ALL_ACTIVE",
      status: "closed",
      monthYear: targetMonth,
      closedBy: String(data.closedBy || data.requestedBy || "Admin").trim(),
      closedAt: nowIso,
      updatedAt: nowIso
    };
    Object.keys(existingRequests).forEach(function(k) {
      const entry = existingRequests[k];
      if (!entry || typeof entry !== "object") return;
      const eMonth = String(entry.monthYear || "").replace(/^'+/, "").trim();
      if (eMonth === targetMonth) {
        entry.status = "closed";
        entry.closedBy = String(data.closedBy || data.requestedBy || "Admin").trim();
        entry.closedAt = nowIso;
        entry.updatedAt = nowIso;
      }
    });
  } else {
    if (data.screenshotRequests && typeof data.screenshotRequests === "object") {
      existingRequests = Object.assign({}, existingRequests, data.screenshotRequests);
    }

function getTodayLocalYmd_() {
  try {
    return Utilities.formatDate(new Date(), Session.getScriptTimeZone() || "Asia/Kolkata", "yyyy-MM-dd");
  } catch (e) {
    return new Date().toISOString().split("T")[0];
  }
}

function pad2Num_(n) {
  return (n < 10 ? "0" : "") + String(n);
}

function parseYmdUtcMs_(ymd) {
  const clean = String(ymd || "").replace(/^'+/, "").trim();
  const m = clean.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!m) return NaN;
  return Date.UTC(Number(m[1]), Number(m[2]) - 1, Number(m[3]));
}

function diffDaysYmd_(fromYmd, toYmd) {
  const a = parseYmdUtcMs_(fromYmd);
  const b = parseYmdUtcMs_(toYmd);
  if (!isFinite(a) || !isFinite(b)) return null;
  return Math.round((b - a) / 86400000);
}

function computeBillDueDateYmd_(billObj, dueDayNum) {
  const dueDay = Math.max(1, Math.min(28, Math.floor(Number(dueDayNum) || 10)));
  const cleanMonth = String((billObj && billObj.MonthYear) || "").replace(/^'+/, "").trim();
  const billDateRaw = String((billObj && billObj.BillDate) || "").replace(/^'+/, "").trim();

  const mm = cleanMonth.match(/^(\d{4})-(\d{2})$/);
  if (mm) {
    let y = Number(mm[1]);
    let m = Number(mm[2]);
    if (m >= 1 && m <= 12) {
      const sameMonthDue = String(y) + "-" + pad2Num_(m) + "-" + pad2Num_(dueDay);
      if (/^\d{4}-\d{2}-\d{2}$/.test(billDateRaw) && billDateRaw > sameMonthDue) {
        m += 1;
        if (m > 12) {
          m = 1;
          y += 1;
        }
        return String(y) + "-" + pad2Num_(m) + "-" + pad2Num_(dueDay);
      }
      return sameMonthDue;
    }
  }

  const bm = billDateRaw.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (bm) {
    let by = Number(bm[1]);
    let bmo = Number(bm[2]);
    const bd = Number(bm[3]);
    if (bd > dueDay) {
      bmo += 1;
      if (bmo > 12) {
        bmo = 1;
        by += 1;
      }
    }
    return String(by) + "-" + pad2Num_(bmo) + "-" + pad2Num_(dueDay);
  }

  return "";
}

function formatMonthDisplay_(monthYearStr) {
  const clean = String(monthYearStr || "").replace(/^'+/, "").trim();
  const m = clean.match(/^(\d{4})-(\d{2})$/);
  if (!m) return clean || "Current Month";
  const names = [
    "January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December"
  ];
  const idx = Number(m[2]) - 1;
  if (idx >= 0 && idx < 12) {
    return names[idx] + " " + m[1];
  }
  return clean;
}

function formatDateDisplay_(ymd) {
  const clean = String(ymd || "").replace(/^'+/, "").trim();
  const m = clean.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!m) return clean || "--";
  const shortMonths = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  const idx = Number(m[2]) - 1;
  return m[3] + " " + (shortMonths[idx] || m[2]) + " " + m[1];
}

function formatInrNum_(val) {
  const n = Math.round((Number(val) || 0) * 100) / 100;
  return n % 1 === 0 ? String(Math.round(n)) : n.toFixed(2);
}

function isValidEmailStr_(emailVal) {
  const s = String(emailVal || "").trim();
  return Boolean(s && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(s));
}

function readBillEmailNotificationLog_(existingGlobalConfig) {
  const defaultLog = {
    newBillSent: {},
    dueSoonSent: {},
    dueDay: 10,
    reminderDaysBefore: 3,
    lastAutoCheckYmd: ""
  };

  let parsed = null;
  if (existingGlobalConfig && existingGlobalConfig.billEmailNotificationLog && typeof existingGlobalConfig.billEmailNotificationLog === "object") {
    parsed = existingGlobalConfig.billEmailNotificationLog;
  } else {
    const rows = readSheetObjects("GlobalConfig");
    const row = rows.find(function(r) { return normKey(r.ConfigKey) === "billEmailNotificationLog".toLowerCase(); });
    if (row && row.ConfigValueJSON) {
      try {
        parsed = JSON.parse(row.ConfigValueJSON);
      } catch (e) {}
    }
  }

  if (!parsed || typeof parsed !== "object") return defaultLog;
  return {
    newBillSent: (parsed.newBillSent && typeof parsed.newBillSent === "object") ? parsed.newBillSent : {},
    dueSoonSent: (parsed.dueSoonSent && typeof parsed.dueSoonSent === "object") ? parsed.dueSoonSent : {},

    reminderDaysBefore: Math.max(0, Math.min(15, Math.floor(parsed.reminderDaysBefore !== undefined ? Number(parsed.reminderDaysBefore) : 3))),
    lastAutoCheckYmd: String(parsed.lastAutoCheckYmd || "").trim()
  };
}

function trimNotificationLogMap_(mapObj, maxEntries) {
  const keys = Object.keys(mapObj || {});
  if (keys.length <= maxEntries) return mapObj;
  const trimmed = {};
  keys.slice(keys.length - maxEntries).forEach(function(k) {
    trimmed[k] = mapObj[k];
  });
  return trimmed;
}

function saveBillEmailNotificationLog_(logObj) {
  const cleanLog = {
    newBillSent: trimNotificationLogMap_(logObj.newBillSent || {}, 400),
    dueSoonSent: trimNotificationLogMap_(logObj.dueSoonSent || {}, 400),
    dueDay: Math.max(1, Math.min(28, Math.floor(Number(logObj.dueDay) || 10))),
    reminderDaysBefore: Math.max(0, Math.min(15, Math.floor(logObj.reminderDaysBefore !== undefined ? Number(logObj.reminderDaysBefore) : 3))),
    lastAutoCheckYmd: String(logObj.lastAutoCheckYmd || "").trim(),

}

function resolveRenterEmailForBill_(billObj, usersList, rentersList, explicitEmail) {
  const users = Array.isArray(usersList) ? usersList : [];
  const renters = Array.isArray(rentersList) ? rentersList : [];

  const billRenterIdKey = normKey((billObj && billObj.RenterID) || "");
  const billPropKey = normKey((billObj && (billObj.PropertyName || billObj.SnapshotPropertyName)) || "");
  const billFlatKey = normKey((billObj && (billObj.FlatName || billObj.SnapshotFlatName)) || "");
  const billTenantKey = normKey((billObj && (billObj.TenantName || billObj.SnapshotTenantName)) || "");
  let billAadhaar = String((billObj && billObj.TenantAadhaar) || "").replace(/\D/g, "").slice(0, 12);

  let matchedRenter = null;
  if (billRenterIdKey) {
    matchedRenter = renters.find(function(r) { return normKey(r.RenterID) === billRenterIdKey; }) || null;
  }
  if (!matchedRenter && billAadhaar.length === 12) {
    matchedRenter = renters.find(function(r) {
      const rAadh = String(r.IdentityDocNumber || r.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
      return rAadh === billAadhaar;
    }) || null;
  }
  if (!matchedRenter && billPropKey && billFlatKey) {
    matchedRenter = renters.find(function(r) {
      return normKey(r.PropertyName) === billPropKey &&
             normKey(r.FlatName) === billFlatKey &&
             normKey(r.Status) === "active";
    }) || null;
  }
  if (!billAadhaar && matchedRenter) {
    billAadhaar = String(matchedRenter.IdentityDocNumber || matchedRenter.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
  }

  const usersWithEmail = users.filter(function(u) {
    return isValidEmailStr_(u && (u.Email || u.email || u.EmailID));
  });

  const getEmail = function(u) {
    return String((u && (u.Email || u.email || u.EmailID)) || "").trim().toLowerCase();
  };
  const isRenterRole = function(u) {
    return normKey(u && u.Role) === "renter";
  };
  const isActiveAccount = function(u) {
    return normKey((u && (u.Status || u.AccountStatus)) || "Active") !== "blocked";
  };

  // 1. Exact Aadhaar match (prefer Renter role & active account)
  if (billAadhaar.length === 12) {
    const aadhMatches = usersWithEmail.filter(function(u) {
      const uAadh = String(u.AssignedAadhaar || u.AadhaarNumber || "").replace(/\D/g, "").slice(0, 12);
      return uAadh === billAadhaar;
    });
    const bestAadh = aadhMatches.find(function(u) { return isRenterRole(u) && isActiveAccount(u); }) ||
      aadhMatches.find(function(u) { return isRenterRole(u); }) ||
      aadhMatches[0];
    if (bestAadh) return getEmail(bestAadh);
  }

  // 2. Assigned Property + Flat match for Renter account
  if (billPropKey && billFlatKey) {
    const flatMatches = usersWithEmail.filter(function(u) {
      return isRenterRole(u) &&
             normKey(u.AssignedProperty) === billPropKey &&
             normKey(u.AssignedFlat) === billFlatKey;
    });
    const bestFlat = flatMatches.find(function(u) { return isActiveAccount(u); }) || flatMatches[0];
    if (bestFlat) return getEmail(bestFlat);
  }

  // 3. FullName match for Renter account
  if (billTenantKey) {
    const nameMatches = usersWithEmail.filter(function(u) {
      return isRenterRole(u) && normKey(u.FullName) === billTenantKey;
    });
    const bestName = nameMatches.find(function(u) { return isActiveAccount(u); }) || nameMatches[0];
    if (bestName) return getEmail(bestName);
  }

  // 4. Explicit email fallback passed from client context
  if (isValidEmailStr_(explicitEmail)) {
    return String(explicitEmail).trim().toLowerCase();
  }

  return "";
}

function checkAndSendApproachingDueDateEmails(opts, preloadedContext) {
  const options = opts || {};
  const ctx = preloadedContext || {};
  const todayYmd = getTodayLocalYmd_();
  const emailLog = readBillEmailNotificationLog_(ctx.globalConfig);

  if (options.dueDay !== undefined) {
    const reqDueDay = Math.floor(Number(options.dueDay) || 0);
    if (reqDueDay >= 1 && reqDueDay <= 28) {
      emailLog.dueDay = reqDueDay;
    }
  }
  const dueDay = emailLog.dueDay || 10;
  const daysBefore = (options.daysBefore !== undefined && !isNaN(Number(options.daysBefore)))
    ? Math.max(0, Math.min(15, Math.floor(Number(options.daysBefore))))
    : (emailLog.reminderDaysBefore !== undefined ? emailLog.reminderDaysBefore : 3);

  const bills = Array.isArray(ctx.bills) ? ctx.bills : readSheetObjects("Bills");
  const users = Array.isArray(ctx.users) ? ctx.users : readSheetObjects("Users");
  const renters = Array.isArray(ctx.renters) ? ctx.renters : readSheetObjects("Renters");
  const flats = Array.isArray(ctx.flats) ? ctx.flats : readSheetObjects("Flats");
  const props = Array.isArray(ctx.properties) ? ctx.properties : readSheetObjects("Properties");

  let sentCount = 0;
  let checkedCount = 0;
  let logDirty = false;
  const sentDetails = [];
  const nowIso = new Date().toISOString();

  bills.forEach(function(bill) {
    const bId = String((bill && bill.BillID) || "").trim();
    if (!bId) return;

    const totalBilled = Number(bill.TotalBilled) || 0;
    const paidAmt = Number(bill.PaidAmount) || 0;
    const pendingAmt = Math.max(0, totalBilled - paidAmt);
    const statusLower = normKey(bill.Status || "");
    if (pendingAmt <= 0 || statusLower.indexOf("paid") !== -1 || statusLower.indexOf("overpaid") !== -1) {
      return;
    }

    checkedCount++;
    const dueDateYmd = computeBillDueDateYmd_(bill, dueDay);
    if (!dueDateYmd) return;

    const daysLeft = diffDaysYmd_(todayYmd, dueDateYmd);
    if (daysLeft === null || daysLeft < 0 || daysLeft > daysBefore) {
      return;
    }

    // Avoid sending both New Bill and Due Soon emails on the very same day a new bill was created
    const newBillEntry = emailLog.newBillSent[bId];
    if (newBillEntry && newBillEntry.ymd === todayYmd) {
      return;
    }

    const dedupeKey = bId + "__" + dueDateYmd;
    if (emailLog.dueSoonSent[dedupeKey]) {
      return;
    }

    const renterEmail = resolveRenterEmailForBill_(bill, users, renters, "");
    if (!renterEmail) return;

    try {
      const emailContent = buildBillNotificationEmail_(
        "DUE_APPROACHING",
        bill,
        dueDateYmd,
        daysLeft,
        flats,
        props
      );
      MailApp.sendEmail({
        to: renterEmail,
        subject: emailContent.subject,
        body: emailContent.textBody,
        htmlBody: emailContent.htmlBody
      });
      emailLog.dueSoonSent[dedupeKey] = {
        at: nowIso,
        ymd: todayYmd,
        email: renterEmail,
        dueDate: dueDateYmd,
        daysLeft: daysLeft,
        pendingAmount: pendingAmt
      };
      logDirty = true;
      sentCount++;
      sentDetails.push({
        billId: bId,
        email: renterEmail,
        dueDate: dueDateYmd,
        daysLeft: daysLeft,
        pendingAmount: pendingAmt
      });
    } catch (sendErr) {}
  });

  if (emailLog.lastAutoCheckYmd !== todayYmd) {
    emailLog.lastAutoCheckYmd = todayYmd;
    logDirty = true;
  }
  if (logDirty) {
    saveBillEmailNotificationLog_(emailLog);
  }
  try {
    CacheService.getScriptCache().put("BILL_DUE_AUTO_CHECK_" + todayYmd, "1", 21600);
  } catch (eCache) {}

  return {
    status: "success",
    today: todayYmd,
    dueDay: dueDay,
    reminderDaysBefore: daysBefore,
    checkedCount: checkedCount,
    sentCount: sentCount,
    details: sentDetails
  };
}

function maybeRunThrottledBillDueEmailCheck_(globalConfig, users, renters, bills, flats, properties) {
  const todayYmd = getTodayLocalYmd_();
  try {
    if (CacheService.getScriptCache().get("BILL_DUE_AUTO_CHECK_" + todayYmd) === "1") {
      return;
    }
  } catch (e1) {}

  const existingLog = readBillEmailNotificationLog_(globalConfig);
  if (existingLog.lastAutoCheckYmd === todayYmd) {
    try {
      CacheService.getScriptCache().put("BILL_DUE_AUTO_CHECK_" + todayYmd, "1", 21600);
    } catch (e2) {}
    return;
  }

  checkAndSendApproachingDueDateEmails(
    { source: "auto_sync" },
    {
      globalConfig: globalConfig,
      users: users,
      renters: renters,
      bills: bills,
      flats: flats,
      properties: properties
    }
  );
}

/**
 * Time-Driven Trigger Entry Point: Run daily in Apps Script to automatically email
 * renters whose bill due date is approaching (within 3 days of due date).
 */
function runDailyBillDueReminderTrigger() {
  ensureAllSheetsAndSchemas();
  return checkAndSendApproachingDueDateEmails({ source: "daily_trigger" });
}

/**
 * One-Click Setup Helper: Run once in Apps Script Editor or via Spreadsheet menu
 * ("Home Management Portal" -> "Enable Daily Bill Due Reminder Emails") to install
 * an automatic daily 9:00 AM background trigger.
 */
function installDailyBillDueReminderTrigger() {
  ensureAllSheetsAndSchemas();
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(function(t) {
    if (t.getHandlerFunction && t.getHandlerFunction() === "runDailyBillDueReminderTrigger") {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger("runDailyBillDueReminderTrigger")
    .timeBased()
    .everyDays(1)
    .atHour(9)
    .create();
  return {
    status: "success",
    message: "Daily bill due-date reminder email trigger installed (runs every day at 9 AM)."
  };
}

function handleDeletePaymentProof(data) {
  if (data.requesterRole && normKey(data.requesterRole) !== "admin") {
    return { status: "error", message: "Permission Denied: Only Admin can delete UTR payment proofs." };
  }
  const proofId = String(data.proofId || data.id || data.ProofID || "").trim();
  if (!proofId) {
    return { status: "error", message: "Missing ProofID." };
  }

  const nowIso = new Date().toISOString();
  const driveIdsToTrash = {};
  const urlsToMatch = {};

  const collectUrlAndId = function(rawVal) {
    const s = String(rawVal || "").trim();
    if (!s) return;
    urlsToMatch[s] = true;
    const fId = extractDriveFileIdFromUrl_(s);
    if (fId) driveIdsToTrash[fId] = true;
  };

  collectUrlAndId(data.screenshotUrl);
  collectUrlAndId(data.imageUrl);
  collectUrlAndId(data.proofImageUrl);
  if (data.driveFileId) driveIdsToTrash[String(data.driveFileId).trim()] = true;

  // Read existing sheet rows so we also catch ScreenshotURL stored only in sheet
  const existingSheetRows = readSheetObjects("PaymentProofs");
  let targetRowMeta = null;
  const remainingSheetRows = [];
  existingSheetRows.forEach(function(r) {
    const rId = String(r.ProofID || "").trim();
    if (rId === proofId) {
      targetRowMeta = r;
      collectUrlAndId(r.ScreenshotURL);
    } else if (rId || r.UTRNumber || r.AmountPaid || r.ScreenshotURL) {
      remainingSheetRows.push(r);
    }
  });
  replaceAllSheetRows("PaymentProofs", remainingSheetRows);

  // Sync GlobalConfig.paymentProofs
  const rawCfgRows = readSheetObjects("GlobalConfig");
  let updatedProofsList = null;
  if (Array.isArray(data.updatedPaymentProofs)) {
    updatedProofsList = data.updatedPaymentProofs;
  } else {
    const foundProofsCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "paymentproofs"; });
    if (foundProofsCfg && foundProofsCfg.ConfigValueJSON) {
      try {
        const parsed = JSON.parse(foundProofsCfg.ConfigValueJSON);
        if (Array.isArray(parsed)) {
          parsed.forEach(function(p) {
            if (p && String(p.id || p.ProofID || "").trim() === proofId) {
              collectUrlAndId(p.screenshotUrl || p.ScreenshotURL || p.imageUrl || p.proofImageUrl);
              if (p.driveFileId) driveIdsToTrash[String(p.driveFileId).trim()] = true;
            }
          });
          updatedProofsList = parsed.filter(function(p) {
            return p && String(p.id || p.ProofID || "").trim() !== proofId;
          });
        }
      } catch (e) {}
    }
  }
  if (updatedProofsList !== null) {
    handleSaveConfigItem("paymentProofs", updatedProofsList);
  }

  // Trash Drive files associated with this UTR proof
  try {
    deletePhotoFromDriveFolderBySlot("PAYMENT_PROOF__" + sanitizeDriveFilePart(proofId));
  } catch (e) {}

  let trashedDriveCount = 0;
  Object.keys(driveIdsToTrash).forEach(function(fId) {
    if (!fId) return;
    try {
      const f = DriveApp.getFileById(fId);
      if (f && !f.isTrashed()) {
        f.setTrashed(true);
        trashedDriveCount++;
      }
    } catch (e) {}
  });

  // Also clean any matching payment screenshot entry in GlobalConfig.renterMonthlyPhotos & Renters.Photo2URL
  const deletedFileIdList = Object.keys(driveIdsToTrash);
  const foundPhotosCfg = rawCfgRows.find(function(r) { return normKey(r.ConfigKey) === "rentermonthlyphotos"; });
  if (foundPhotosCfg && foundPhotosCfg.ConfigValueJSON && (deletedFileIdList.length > 0 || Object.keys(urlsToMatch).length > 0)) {
    try {
      const photosMap = JSON.parse(foundPhotosCfg.ConfigValueJSON) || {};
      let mapChanged = false;
      Object.keys(photosMap).forEach(function(k) {
        const entry = photosMap[k];
        if (!entry || typeof entry !== "object") return;
        const eFid = String(entry.driveFileId || entry.fileId || extractDriveFileIdFromUrl_(entry.imageUrl || entry.previewUrl || "")).trim();
        const eUrl = String(entry.imageUrl || entry.previewUrl || "").trim();
        if ((eFid && driveIdsToTrash[eFid]) || (eUrl && urlsToMatch[eUrl])) {
          if (entry.slotKey) {
            try { deletePhotoFromDriveFolderBySlot(entry.slotKey); } catch (e) {}
          }
          delete photosMap[k];
          mapChanged = true;
        }
      });
      if (mapChanged) {
        handleSaveConfigItem("renterMonthlyPhotos", photosMap);
      }
    } catch (e) {}
  }

  if (deletedFileIdList.length > 0 || Object.keys(urlsToMatch).length > 0) {
    updateRowsByMatcher(
      "Renters",
      function(r) {
        const p2 = String(r.Photo2URL || "").trim();
        if (!p2) return false;
        const p2Fid = extractDriveFileIdFromUrl_(p2);
        return Boolean((p2Fid && driveIdsToTrash[p2Fid]) || urlsToMatch[p2]);
      },
      function() {
        return { Photo2URL: "", UpdatedAt: nowIso };
      }
    );
  }

  return {
    status: "success",
    proofId: proofId,
    trashedDriveFiles: trashedDriveCount,
    message: "UTR record and Google Drive screenshot deleted permanently."
  };
}

 */
function handleSyncAllData(data) {
  const nowIso = new Date().toISOString();
  const props = Array.isArray(data.properties) ? data.properties : [];
  const flats = Array.isArray(data.flats) ? data.flats : [];
  const renters = Array.isArray(data.renters) ? data.renters : [];
  const bills = Array.isArray(data.bills) ? data.bills : [];
  const users = Array.isArray(data.users) ? data.users : [];
  const fds = Array.isArray(data.fdRecords) ? data.fdRecords : [];
  const khets = Array.isArray(data.khetRecords) ? data.khetRecords : [];
  const kharchs = Array.isArray(data.kharchRecords) ? data.kharchRecords : [];

  props.forEach(function(p) {
    if (p && p.PropertyName) {
      upsertRowByMatcher("Properties", function(r) { return normKey(r.PropertyName) === normKey(p.PropertyName); }, Object.assign({}, p, { UpdatedAt: nowIso }));
    }
  });

  flats.forEach(function(f) {
    if (f && f.PropertyName && f.FlatName) {
      upsertRowByMatcher("Flats", function(r) { return normKey(r.PropertyName) === normKey(f.PropertyName) && normKey(r.FlatName) === normKey(f.FlatName); }, Object.assign({}, f, { UpdatedAt: nowIso }));
    }
  });

  renters.forEach(function(r) {
    if (r && r.RenterID) {
      const copy = Object.assign({}, r, {
        UpdatedAt: nowIso
      });
      if (Array.isArray(copy.FamilyMembersList)) {
        copy.FamilyMembersList = JSON.stringify(copy.FamilyMembersList);
      }
      upsertRowByMatcher("Renters", function(row) { return normKey(row.RenterID) === normKey(r.RenterID); }, copy);
    }
  });

  bills.forEach(function(b) {
    if (b && b.BillID) {
      upsertRowByMatcher("Bills", function(row) { return normKey(row.BillID) === normKey(b.BillID); }, Object.assign({}, b, { UpdatedAt: nowIso }));
    }
  });

  if (users.length > 0) {
    handleSaveUsers({ users: users });
  }

  fds.forEach(function(fd) {
